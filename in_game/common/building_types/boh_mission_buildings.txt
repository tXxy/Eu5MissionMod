# Unique Bohemian buildings. Like vanilla's event-only buildings they cannot be built by hand:
# the matching Bohemia mission constructs them.

# Karlštejn Castle, built by Charles IV to guard the crown jewels. Placed at Beroun.
# Built by mission_boh_crown_of_saint_wenceslas.
boh_karlstejn_castle = {
	audio_tier = 3
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1

	category = cultural_category

	employment_size = 1

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	country_potential = { has_variable = boh_unlocked_karlstejn }
	allow = { always = no }

	expensive = yes

	build_time = large_cultural_building_time

	modifier = {
		local_defensive = 0.5
		local_cultural_tradition = 0.2
		local_crown_estate_power = 1.0
	}

	possible_production_methods = {
		capital_building_maintenance
	}
	construction_demand = capital_building_construction
}
