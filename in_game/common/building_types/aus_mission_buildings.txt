# Unique Austrian buildings. Like vanilla's event-only buildings they cannot be built by hand:
# the matching Austria mission constructs them.

# The Hofburg, the residence of the Habsburgs in Vienna.
# Built by mission_aus_hofburg.
aus_hofburg = {
	audio_tier = 4
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1

	category = cultural_category

	employment_size = 1

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	country_potential = { has_variable = aus_unlocked_hofburg }
	allow = { always = no }

	expensive = yes

	build_time = large_cultural_building_time

	modifier = {
		local_crown_estate_power = 1.0
		local_cultural_tradition = 0.2
	}

	capital_country_modifier = {
		monthly_legitimacy = 0.05
		diplomatic_capacity_modifier = 0.1
	}

	possible_production_methods = {
		capital_building_maintenance
	}
	construction_demand = capital_building_construction
}
