# Herrenstand and Ritterstand: the lords and knights who sit in the Austrian Landtage.
# Unlocked by mission_aus_landtage.

	estate = nobles_estate

	potential = { has_or_had_tag = HAB }

	allow = {
		has_unlocked_estate_privilege_trigger = {
			type = aus_herren_und_ritterstand
		}
	}

	country_modifier = {
		nobles_estate_target_satisfaction = medium_privilege_target_satisfaction
		global_nobles_estate_power = 0.1
		global_manpower_modifier = 0.05
		army_heavy_cavalry_power = 0.03
	}
}
