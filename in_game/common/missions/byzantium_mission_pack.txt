byzantium_grand_mission_pack = {
	icon = generic_empire
	chance = 3600
	repeatable = no
	player_playstyle = military

	visible = {
		game_has_missions_enabled = yes
		tag = "BYZ"
	}

	on_start = {
		set_variable = {
			name = byz_target_navy_size
			value = 20
		}
		set_variable = {
			name = byz_silk_income_target
			value = 15
		}
	}

	on_completion = {
		add_prestige = 100
		add_legitimacy = 50
		remove_variable = byz_target_navy_size
		remove_variable = byz_silk_income_target
	}

	# =========================================================
	# === PILLAR 1: THE PURPLE PHOENIX (Internal & Survival) ===
	# =========================================================

	mission_secure_theodosian_walls = {
		icon = defensive_army
		requires = { }
		enabled = {
			location:constantinople = {
				has_building_with_at_least_one_level = castle
				control >= 90
				levy_size >= 3000 
				# Ensure walls are maintained
				maintenance_cost_paid = yes
			}
			manpower >= 4000
			stability >= 40
		}
		duration = 0
		on_completion = {
			add_prestige = 15
			location:constantinople = {
				add_location_modifier = {
					modifier = theodosian_walls_restored
					years = 25
					mode = add
				}
				change_defensiveness = 0.25
			}
			trigger_event_silently = byz_flavor.1
		}
	}

	mission_curtail_the_dynatoi = {
		icon = nobility
		requires = { mission_secure_theodosian_walls }
		enabled = {
			"estate_power(estate_type:nobles_estate)" <= 0.45
			"estate_power(estate_type:crown_estate)" >= 0.35
			legitimacy >= 60
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = imperial_authority_asserted
				years = 15
				mode = add
			}
			add_estate_satisfaction = {
				type = estate_type:nobles_estate
				value = estate_satisfaction_weak_penalty
			}
			add_gold = 150
			trigger_event_non_silently = byz_flavor.2
		}
	}

	mission_reform_pronoia_system = {
		icon = manpower_recovery_speed
		requires = { mission_curtail_the_dynatoi }
		enabled = {
			has_country_modifier = reformed_pronoia_system # Set via event or decision
			army_professionalism >= 0.2
			army_size_percentage >= 0.9
		}
		duration = 0
		on_completion = {
			add_manpower_recovery_speed = 0.15
			add_max_manpower_modifier = 0.2
		}
	}

	mission_restore_the_senate = {
		icon = government_reforms
		requires = { mission_reform_pronoia_system }
		enabled = {
			capital = {
				has_building_with_at_least_one_level = court
				development >= 55
			}
			average_country_literacy >= 40
			stability >= 75
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = byzantine_senate_restored
				years = -1
				mode = add
			}
			add_administrative_efficiency = 0.05
			trigger_event_silently = byz_flavor.3
		}
	}

	# =========================================================
	# === PILLAR 2: THE GOLDEN HORN (Economy & Navy) ===
	# =========================================================

	mission_repair_golden_horn_chain = {
		icon = ship_building_advance
		requires = { mission_secure_theodosian_walls }

	mission_rebuild_imperial_navy = {
		icon = maritime_advance_age_2
		requires = { mission_repair_golden_horn_chain }
		enabled = {
			navy_size >= var:byz_target_navy_size

		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = thalassocracy_ambitions
				years = 15
				mode = add
			}

		}
	}

	mission_revitalize_silk_industry = {
		icon = merchants_and_trade
		requires = { mission_curtail_the_dynatoi }
		enabled = {
			# Check for silk production
			any_owned_location = {
				count >= 3
				raw_material = goods:silk
				has_building_with_at_least_one_level = textile_manufactory
			}
			# Check income from silk
			produced_in_country = {
				goods = goods:silk
				value >= var:byz_silk_income_target
			}
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = imperial_silk_monopoly
				years = 20
				mode = add
			}
			add_gold = {
				value = monthly_income
				multiply = 12
			}
			trigger_event_non_silently = byz_flavor.4
		}
	}

	mission_dominate_black_sea_trade = {
		icon = trade_efficiency
		requires = { mission_rebuild_imperial_navy mission_revitalize_silk_industry }
		enabled = {
			# Control key ports
			location:kaffa = { owner ?= root }
			location:trebizond = { owner ?= root }
			
			capital.market = {
				"merchant_power_in_market(root)" >= 65
			}
		}
		duration = 0
		on_completion = {
			add_trade_efficiency = 0.15
			add_mercantilism = 0.05
			capital = {
				change_prosperity = prosperity_strong_bonus
			}
		}
	}
