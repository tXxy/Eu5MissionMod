netherlands_formation_mission_pack = {
	icon = generic_capital_economy
	repeatable = no
	player_playstyle = diplomatic

	visible = {
		game_has_missions_enabled = yes
		OR = {
			tag = "HOL"
			tag = "NET"
		}
		NOT = { has_country_flag = netherlands_formation_completed }
	}

	enabled = {
		exists = capital
		development >= 100
	}
	
	chance = 3600

	on_start = {
		set_variable = {
			name = mission_target_development
			value = {
				value = development
				add = 50
			}
		}
	}

	on_completion = {
		set_country_flag = netherlands_formation_completed
		add_prestige = 100
		remove_variable = mission_target_development
	}

	on_abort = {
		remove_variable = mission_target_development
	}

	# === UNIFICATION PREREQUISITES ===
	
	mission_unite_low_countries = {
		icon = diplomatic_influence
		requires = { }
		
		enabled = {
			custom_tooltip = {
				text = control_dutch_culture_provinces_tt
				num_of_provinces_with_dutch_culture >= 8
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_cultural_unity
				years = 20
			}
			add_prestige = 25
		}
	}

	mission_secure_independence = {
		icon = glorious_arms
		requires = { }
		
		enabled = {
			NOT = { is_subject = yes }
			liberty_desire <= 0
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_independence
				years = 15
			}
			add_army_tradition = 25
		}
	}

	# === STATE BUILDING ===
	
	mission_establish_republic = {
		icon = bill_of_rights
		requires = { mission_secure_independence mission_unite_low_countries }
		
		enabled = {
			government = republic
			stability >= 1
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_republican_traditions
				years = -1
			}
			add_government_reform_progress = 50
		}
	}

	mission_economic_supremacy = {
		icon = wealth_renaissance
		requires = { mission_establish_republic }
		
		enabled = {
			monthly_trade_income >= 75
			development >= var:mission_target_development
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_economic_hegemony
				years = 25
			}
			add_gold = 500
		}
	}

	mission_naval_supremacy = {
		icon = global_trade_routes_advance
		requires = { mission_establish_republic }
		
		enabled = {
			navy_size >= 100
			num_of_ports >= 12
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_naval_dominance
				years = 25
			}
			add_navy_tradition = 35
		}
	}

	# === GLOBAL AMBITIONS ===
	
	mission_colonial_empire = {
		icon = colonial_heritage
		requires = { mission_economic_supremacy }
		
		enabled = {
			num_colonies >= 5
			num_trade_companies >= 2
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_global_empire
				years = 30
			}
			add_colonist = 1
		}
	}

	mission_dutch_golden_age = {
		icon = cultural_acceptance_advance
		requires = { mission_naval_supremacy mission_colonial_empire }
		final = yes
		
		enabled = {
			prestige >= 75
			technology_group = western
			any_estate = {
				loyalty >= 60
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = netherlands_golden_age
				years = -1
			}
			add_prestige = 150
			add_splendor = 200
			custom_tooltip = netherlands_golden_age_tt
		}
	}
}
