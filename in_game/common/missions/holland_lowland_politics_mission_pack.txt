# holland_lowland_politics_mission_pack
holland_lowland_politics_mission_pack = {
    icon = diplomacy
    repeatable = no
    player_playstyle = diplomatic

    visible = {
        game_has_missions_enabled = yes
        tag = HOL
    }

    chance = 3600

    # Secure Local Alliances
    mission_secure_alliances = {
        icon = diplomatic_influence
        enabled = {
            num_alliances >= 1
        }
        duration = 0
        on_completion = {
            add_diplomatic_reputation = 1
        }
    }

    # Balance Against Burgundy
    mission_against_burgundy = {
        icon = diplomacy_power
        requires = { mission_secure_alliances }
        enabled = {
            any_neighbor_country = { tag = BUR }
        }
        duration = 0
        on_completion = {
            add_spy_network_speed = 0.2
            trigger_event_non_silently = holland_politics_events.1
        }
    }

    # Lowland Influence
    mission_lowland_influence = {
        icon = cultural_acceptance_advance
        requires = { mission_against_burgundy }
        enabled = {
            diplomatic_influence > 15
        }
        duration = 365
        modifier_while_progressing = {
            global_relations_improve_modifier = 0.08
        }
        on_completion = {
            add_country_modifier = {
                modifier = holland_lowland_diplomacy_modifier
                years = 20
            }
        }
    }

    # Dutch Autonomy
    mission_dutch_autonomy = {
        icon = political_influence
        requires = { mission_lowland_influence }
        enabled = {
            stability > 0
            legitimacy > 40
        }
        duration = 0
        on_completion = {
            add_government_reform_progress = 0.20
        }
    }

    # Final: Foundation of the Dutch States
    mission_dutch_states = {
        icon = correct_box_advance_renaissance
        final = yes
        requires = { mission_dutch_autonomy }
        enabled = {
            capital = { development > 25 }
            any_owned_location = { culture = dutch }
        }
        duration = 0
        on_completion = {
            add_global_tax_modifier = 0.08
            add_global_manpower = 0.08
            trigger_event_non_silently = holland_politics_events.10
        }
    }
}
