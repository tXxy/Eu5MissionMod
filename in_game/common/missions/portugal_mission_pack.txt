portugal_mission_pack = {
	icon = generic_maritime_expansion
	chance = 3600
	repeatable = no
	player_playstyle = diplomatic
	visible = {
		game_has_missions_enabled = yes
		tag = "POR"
	}
	on_start = {
		set_variable = {
			name = mission_target_income
			value = 20
		}
	}
	on_completion = {
		add_prestige = 30
		add_navy_tradition = 25
		remove_variable = mission_target_income
	}
	on_abort = {
		remove_variable = mission_target_income
	}

	# === PORTUGUESE CONSOLIDATION PATH ===
	mission_secure_portuguese_borders = {
		icon = defensive_army
		requires = { }
		enabled = {
			region:portugal_region = {
				locations_owned_by_fraction = {
					target = root
					value >= 0.95
				}
			}
			stability >= 1
			estate_satisfaction:nobles_estate >= 0.4
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = portuguese_consolidation
				years = 10
				mode = add
			}
			add_prestige = 15
			trigger_event_non_silently = portugal_mission_events.1
		}
	}

	mission_develop_lisbon = {
		icon = cultural_acceptance_advance
		requires = { mission_secure_portuguese_borders }
		enabled = {
			capital = {
				development >= 25
				total_building_levels >= 10
				has_building_with_at_least_one_level = marketplace
			}
			monthly_income >= var:mission_target_income
		}
		duration = 0
		on_completion = {
			add_gold = {
				value = monthly_income
				multiply = 12
			}
			capital = {
				add_development = 5
				add_location_modifier = {
					modifier = lisbon_commercial_hub_location
					years = 10
					mode = add
				}
			}
			add_country_modifier = {
				modifier = lisbon_commercial_hub
				years = 10
				mode = add
			}
		}
	}

	mission_strengthen_monarchy = {
		icon = court_accounting
		requires = { mission_secure_portuguese_borders }
		enabled = {
			crown_estate_power >= 0.4
			prestige >= 25
			ruler = {
				legitimacy >= 60
				is_female = no
			}
			"estate(estate_type:nobles_estate)" = {
				num_privileges >= 2
			}
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = avis_dynasty_strength
				years = 15
				mode = add
			}
			add_legitimacy = 25
			add_estate_satisfaction = {
				type = estate_type:nobles_estate
				value = estate_satisfaction_mild_bonus
			}
		}
	}

	# === MARITIME EXPANSION PATH ===
	mission_build_portuguese_navy = {
		icon = ship_building_advance
		requires = { mission_develop_lisbon }
		enabled = {
			navy_size >= 20
			num_of_ports >= 4
			total_effective_building_levels:wharf >= 3
			total_effective_building_levels:dry_dock >= 2
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = portuguese_naval_tradition
				years = 10
				mode = add
			}
			add_navy_tradition = 15
			ordered_owned_location = {
				limit = {
					is_coastal = yes
					has_building_with_at_least_one_level = dry_dock
				}
				order_by = development
				max = 1
				change_development = development_mild_bonus
			}
		}
	}

	mission_explore_african_coast = {

		icon = diplomatic_influence
		requires = { mission_secure_portuguese_borders }
		enabled = {
			OR = {
				country_exists = c:ENG
				country_exists = c:GBR
			}
			OR = {
				c:ENG = { 
					is_allied_with = { target = root }
					has_royal_marriage = { target = root }
					trust = { target = root value >= 60 }
				}
				c:GBR = { 
					is_allied_with = { target = root }
					has_royal_marriage = { target = root }
					trust = { target = root value >= 60 }
				}
			}
			diplomatic_reputation >= 2
		}
		duration = 0
		on_completion = {
			add_country_modifier = {
				modifier = anglo_portuguese_alliance
				years = 15
				mode = add
			}
			add_diplomatic_reputation = 1
			OR = {
				c:ENG = { add_trust = { modifier = trust_mission_trusted_ties target = root } }
				c:GBR = { add_trust = { modifier = trust_mission_trusted_ties target = root } }
			}
		}
	}
