portugal_mission_pack = {
	icon = generic_maritime_expansion
	chance = 3600
	repeatable = no
	player_playstyle = diplomatic

	visible = {
		game_has_missions_enabled = yes
		tag = "POR"
	}

	on_start = {
		set_variable = {
			name = mission_target_income
			value = 20
		}
	}

	on_completion = {
		add_prestige = 30
		add_navy_tradition = 25
		remove_variable = mission_target_income
	}

	on_abort = {
		remove_variable = mission_target_income
	}

	# === PORTUGUESE CONSOLIDATION PATH ===
	
	mission_secure_portuguese_borders = {
		icon = defensive_army
		requires = { }
		
		enabled = {
			any_owned_location = {
				region = region:portugal_region
				count >= 12
			}
			stability >= 1
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_consolidation
				years = 10
			}
			add_prestige = 15
			trigger_event_non_silently = portugal_mission_events.1
		}
	}

	mission_develop_lisbon = {
		icon = cultural_acceptance_advance
		requires = { mission_secure_portuguese_borders }
		
		enabled = {
			capital = {
				development >= 20
			}
			monthly_income >= 15
		}
		
		duration = 0

		on_completion = {
			add_gold = 150
			capital = {
				add_development = 5
			}
			add_country_modifier = {
				modifier = lisbon_commercial_hub
				years = 10
			}
		}
	}

	mission_strengthen_monarchy = {
		icon = court_accounting
		requires = { mission_secure_portuguese_borders }
		
		enabled = {
			crown_estate_power >= 0.4
			prestige >= 25
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = avis_dynasty_strength
				years = 15
			}
			add_legitimacy = 25
		}
	}

	# === MARITIME EXPANSION PATH ===
	
	mission_build_portuguese_navy = {
		icon = ship_building_advance
		requires = { mission_develop_lisbon }
		
		enabled = {
			navy_size >= 15
			num_of_ports >= 3
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_naval_tradition
				years = 10
			}
			add_navy_tradition = 15
		}
	}

	mission_explore_african_coast = {
		icon = route_to_the_indies_advance
		requires = { mission_build_portuguese_navy }
		
		enabled = {
			has_discovered_region = region:west_africa
			num_of_ports >= 4
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_exploration_spirit
				years = 15
			}
			trigger_event_non_silently = portugal_mission_events.2
		}
	}

	mission_establish_trading_posts = {
		icon = merchants_and_trade
		requires = { mission_explore_african_coast }
		
		enabled = {
			any_owned_location = {
				continent = continent:africa
				count >= 2
			}
			monthly_income >= 10
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_trading_network
				years = 10
			}
			add_gold = 200
		}
	}

	# === NORTH AFRICAN EXPANSION PATH ===
	
	mission_ceuta_expedition = {
		icon = glorious_arms
		requires = { mission_build_portuguese_navy }
		
		enabled = {
			location:Ceuta = {
				owner = root
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_moroccan_presence
				years = 10
			}
			add_prestige = 20
			trigger_event_silently = portugal_mission_events.10
		}
	}

	mission_secure_moroccan_coast = {
		icon = maritime_advance_age_2
		requires = { mission_ceuta_expedition }
		
		enabled = {
			any_owned_location = {
				region = region:morocco_region
				count >= 3
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_north_african_foothold
				years = 15
			}
			add_manpower = 3000
		}
	}

	# === ATLANTIC EXPLORATION PATH ===
	
	mission_discover_azores = {
		icon = global_trade_routes_advance
		requires = { mission_explore_african_coast }
		
		enabled = {
			has_discovered_region = region:azores
			any_owned_location = {
				region = region:azores
				count >= 1
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_atlantic_knowledge
				years = 10
			}
			trigger_event_silently = portugal_mission_events.3
		}
	}

	mission_colonize_madeira = {
		icon = colonial_heritage
		requires = { mission_discover_azores }
		
		enabled = {
			any_owned_location = {
				region = region:madeira
				count >= 1
			}
			total_development >= 200
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_island_colonies
				years = 15
			}
			add_population_growth = 0.05
		}
	}

	# === ECONOMIC DEVELOPMENT PATH ===
	
	mission_develop_port_wine = {
		icon = expanded_supply_trains
		requires = { mission_secure_portuguese_borders }
		
		enabled = {
			any_owned_location = {
				has_building = vineyard
				count >= 3
			}
			monthly_income >= 15
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_wine_production
				years = 10
			}
			add_gold = 100
		}
	}

	mission_establish_fishing_industry = {
		icon = maritime_advance_age_2
		requires = { mission_develop_port_wine }
		
		enabled = {
			any_owned_location = {
				has_building = fishery
				count >= 4
			}
			monthly_income >= 20
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_fishing_industry
				years = 10
			}
			add_manpower = 2000
		}
	}

	# === DIPLOMATIC PATH ===
	
	mission_english_alliance = {
		icon = diplomatic_influence
		requires = { mission_secure_portuguese_borders }
		
		enabled = {
			has_ally = ENG
			has_royal_marriage = ENG
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = anglo_portuguese_alliance
				years = 15
			}
			add_diplomatic_reputation = 1
		}
	}

	mission_counter_castilian_influence = {
		icon = improve_relation_impact_renaissance
		requires = { mission_english_alliance }
		
		enabled = {
			NOT = { is_subject = CAS }
			NOT = { is_subject = SPA }
			prestige >= 30
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_independence_asserted
				years = 10
			}
			add_prestige = 25
		}
	}

	# === MAJOR ACHIEVEMENTS ===
	
	mission_route_to_indies = {
		icon = route_to_the_indies_advance
		requires = { mission_establish_trading_posts mission_colonize_madeira }
		
		enabled = {
			has_discovered_region = region:india
			num_of_ports >= 6
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_indies_route
				years = 20
			}
			trigger_event_non_silently = portugal_mission_events.11
		}
	}

	mission_portuguese_empire = {
		icon = bill_of_rights
		requires = { 
			mission_route_to_indies 
			mission_secure_moroccan_coast 
			mission_counter_castilian_influence 
		}
		final = yes
		
		enabled = {
			any_owned_location = {
				continent = continent:asia
				count >= 1
			}
			any_owned_location = {
				continent = continent:africa  
				count >= 3
			}
			any_owned_location = {
				continent = continent:south_america
				count >= 1
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_global_empire
				years = -1
			}
			add_prestige = 100
			add_navy_tradition = 50
			trigger_event_non_silently = portugal_mission_events.12
		}
	}
}
