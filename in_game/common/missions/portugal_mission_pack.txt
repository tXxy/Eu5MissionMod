portugal_mission_pack = {
	icon = generic_mission_pack
	chance = 3600
	repeatable = no
	player_playstyle = all_rounder

	visible = {
		game_has_missions_enabled = yes
		tag = "POR"
	}

	on_start = {
		set_variable = {
			name = portugal_mission_progress
			value = 0
		}
	}

	on_completion = {
		add_prestige = 50
		diplomatic_reputation = 2
		remove_variable = portugal_mission_progress
	}

	on_abort = {
		remove_variable = portugal_mission_progress
	}

	# === FOUNDATION AND RECONQUISTA ===
	
	mission_secure_the_throne = {
		icon = court_accounting
		requires = { }
		
		enabled = {
			stability >= 50
			legitimacy >= 60
		}
		
		duration = 0

		on_completion = {
			add_legitimacy = 25
			add_prestige = 15
			trigger_event_non_silently = portugal_mission_events.1
		}
	}

	mission_complete_portuguese_reconquista = {
		icon = glorious_arms
		requires = { mission_secure_the_throne }
		
		enabled = {
			total_development >= 120
			average_control_in_home_region >= 0.75
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_reconquista_complete
				years = 10
			}
			add_army_tradition = 25
			trigger_event_silently = portugal_mission_events.2
		}
	}

	mission_consolidate_the_kingdom = {
		icon = bill_of_rights
		requires = { mission_complete_portuguese_reconquista }
		
		enabled = {
			dominant_culture = culture:portuguese
			average_control_in_home_region >= 0.85
			total_development >= 200
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = consolidated_portuguese_kingdom
				years = 15
			}
			add_stability = 10
		}
	}

	# === NAVAL AND EXPLORATION PATH ===
	
	mission_develop_portuguese_navy = {
		icon = naval_influence
		requires = { mission_consolidate_the_kingdom }
		
		enabled = {
			navy_size >= 25
			navy_tradition >= 20
			monthly_income_trade_and_tax >= 50
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_naval_tradition
				years = 10
			}
			add_navy_tradition = 25
			trigger_event_non_silently = portugal_mission_events.3
		}
	}

	mission_establish_sagres_school = {
		icon = cultural_acceptance_advance
		requires = { mission_develop_portuguese_navy }
		
		enabled = {
			technology_group = western
			num_possible_artists >= 2
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = sagres_navigation_school

				years = -1
			}
			add_country_modifier = {
				modifier = navigation_advancements

				years = 10
			}
			trigger_event_silently = portugal_mission_events.4
		}
	}

	mission_discover_madeira = {
		icon = exploration
		requires = { mission_establish_sagres_school }
		
		enabled = {
			naval_range >= 150
		}
		
		duration = 0

		on_completion = {
			add_gold = 200
			add_prestige = 25
			trigger_event_non_silently = portugal_mission_events.5
		}
	}

	mission_discover_azores = {
		icon = exploration
		requires = { mission_discover_madeira }
		
		enabled = {
			naval_range >= 200
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = atlantic_exploration

				years = 10
			}
			add_navy_tradition = 15
		}
	}

	# === AFRICAN EXPANSION ===
	
	mission_conquer_ceuta = {
		icon = glorious_arms
		requires = { mission_develop_portuguese_navy }
		
		enabled = {
			any_neighbor_country = {
				religion_group = muslim
				is_neighbor_of = ROOT
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = foothold_in_africa

				years = 10
			}
			add_army_tradition = 20
			trigger_event_silently = portugal_mission_events.6
		}
	}

	mission_establish_west_african_trade = {
		icon = trade_company
		requires = { mission_conquer_ceuta }
		
		enabled = {
			num_merchants >= 2
			trade_efficiency >= 0.15
		}
		
		duration = 0

		on_completion = {
			add_gold = 300
			add_country_modifier = {
				modifier = west_african_trade

				years = 10
			}
			trigger_event_non_silently = portugal_mission_events.7
		}
	}

	mission_control_gold_coast = {
		icon = trade_company
		requires = { mission_establish_west_african_trade }
		
		enabled = {
			trade_efficiency >= 0.25
			global_merchant_power >= 25
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = gold_coast_dominance

				years = 15
			}
			add_country_modifier = {
				modifier = gold_trade

				years = 10
			}
		}
	}

	# === INDIAN OCEAN AND ASIA ===
	
	mission_round_the_cape = {
		icon = exploration
		requires = { mission_control_gold_coast }
		
		enabled = {
			naval_range >= 350
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = cape_route_discovered

				years = 10
			}
			add_prestige = 50
			trigger_event_silently = portugal_mission_events.8
		}
	}

	mission_establish_goa = {
		icon = colonial_company
		requires = { mission_round_the_cape }
		
		enabled = {
			colonial_range >= 400
			num_colonies >= 2
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_india

				years = 15
			}
			add_gold = 500
			trigger_event_non_silently = portugal_mission_events.9
		}
	}

	mission_control_spice_trade = {
		icon = trade_company
		requires = { mission_establish_goa }
		
		enabled = {
			trade_income >= 100
			trade_efficiency >= 0.30
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = spice_trade_monopoly

				years = 20
			}
			add_country_modifier = {
				modifier = indian_spice_trade

				years = 15
			}
		}
	}

	mission_establish_macau = {
		icon = colonial_company
		requires = { mission_control_spice_trade }
		
		enabled = {
			num_colonies >= 3
			colonial_range >= 500
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = gateway_to_china

				years = -1
			}
			add_prestige = 75
		}
	}

	# === BRAZIL AND NEW WORLD ===
	
	mission_discover_brazil = {
		icon = exploration
		requires = { mission_round_the_cape }
		
		enabled = {
			naval_range >= 300
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = claim_to_brazil

				years = 10
			}
			trigger_event_silently = portugal_mission_events.10
		}
	}

	mission_colonize_brazil = {
		icon = colonial_company
		requires = { mission_discover_brazil }
		
		enabled = {
			num_colonies >= 5
			colonial_migration_size_modifier >= 0.20
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = brazilian_colony_established

				years = 15
			}
			add_country_modifier = {
				modifier = madeira_settlement

				years = 10
			}
		}
	}

	mission_develop_brazilian_trade = {
		icon = trade_company
		requires = { mission_colonize_brazil }
		
		enabled = {
			trade_income >= 150
			monthly_income_from_colonies >= 100
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = brazilian_trade_boom

				years = 20
			}
			add_gold = 750
		}
	}

	# === GLOBAL EMPIRE FINALE ===
	
	mission_build_global_empire = {
		icon = imperial_authority
		requires = { mission_establish_macau mission_develop_brazilian_trade }
		
		enabled = {
			num_colonies >= 10
			total_development >= 500
			monthly_income_trade_and_tax >= 200
			navy_size >= 50
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_global_empire

				years = 25
			}
			add_prestige = 100
			trigger_event_non_silently = portugal_mission_events.11
		}
	}

	mission_portuguese_golden_age = {
		icon = golden_age
		requires = { mission_build_global_empire }
		final = yes
		
		enabled = {
			prestige >= 80
			technology_group = western
			great_power_rank <= 3
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = portuguese_golden_age

				years = 30
			}
			add_gold = 1000
			add_country_modifier = {
				modifier = cartographic_expertise

				years = 10
			}
			custom_tooltip = portuguese_golden_age_tt
		}
	}
}
