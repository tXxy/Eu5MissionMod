ottoman_missions = {
    icon = generic_conquer_province
    repeatable = no
    chance = 3600
    player_playstyle = expansionist
    #ai_will_do = { value = 100 }

    visible = {
        game_has_missions_enabled = yes
        OR = {
            has_or_had_tag = TUR
        }
    }

    on_start = {
        add_country_modifier = {
            modifier = tur_ghazi_tradition
            days = -1
            mode = add_and_extend
        }
    }

    on_abort = {
        remove_country_modifier = tur_ghazi_tradition
    }

    on_completion = {
        add_prestige = 50
        add_legitimacy = 25
    }

    # ==================== PHASE 1: FOUNDATIONS (EARLY GAME) ====================

    mission_tur_consolidate_anatolia = {
        icon = centralization_advance
        #ai_will_do = { value = 100 }
        duration = 0
        requires = { }

        enabled = {
            stability >= 20
            legitimacy >= 50
            area:anatolia_area = {
                any_location_in_area = {
                    OR = {
                        custom_tooltip = {
                            text = is_owner_generic_check
                            owner ?= root
                        }
                        custom_tooltip = {
                            text = is_owner_generic_subject_check
                            owner ?= { is_subject_of = root }
                        }
                    }
                    percent >= 0.7
                }
            }
            "estate_power(estate_type:crown_estate)" >= 0.25
        }

        on_completion = {
            add_legitimacy = 15
            add_country_modifier = {
                modifier = tur_anatolian_consolidation
                years = 20
                mode = add_and_extend
            }
            trigger_event_silently = ottoman_mission_events.1
        }
    }

    mission_tur_develop_bursa = {
        icon = cultural_acceptance_advance
