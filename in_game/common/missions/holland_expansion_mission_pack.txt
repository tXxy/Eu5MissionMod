	# ==================== PHASE 1: HOLLAND CONSOLIDATION (1337-1450) ====================
	
	mission_hol_develop_amsterdam = {
		icon = cultural_acceptance_advance
		requires = { }
		
		enabled = {
			owns = location:amsterdam
			location:amsterdam = {
				development >= 25
				has_building_with_at_least_one_level = marketplace
			}
			monthly_income_trade_and_tax >= 15
		}
		
		duration = 0

		on_completion = {
			location:amsterdam = {
				add_location_modifier = {
					modifier = amsterdam_trading_hub
					years = 25
					mode = add
				}
				change_development = 3
			}
		}
		ai_will_do = { value = 100 }
	}

	mission_hol_consolidate_holland = {
		icon = defensive_army
		requires = { }
		
		enabled = {
			area:holland_area = {
				any_location_in_area = {
					OR = {
						custom_tooltip = {
							text = is_owner_generic_check
							owner ?= root
						}
						custom_tooltip = {
							text = is_owner_generic_subject_check
							owner ?= { is_subject_of = root }
						}
					}
					percent >= 0.75
				}
			}
			stability >= 30
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = hol_holland_consolidated
				years = 20
				mode = add
			}
			add_gold = 150
		}
		ai_will_do = { value = 100 }
	}

	mission_hol_strengthen_burghers = {
