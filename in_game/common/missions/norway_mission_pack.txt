	mission_nor_treaty_of_novgorod = {
		icon = diplomatic_influence
		requires = { mission_nor_finnmark_frontier }

		enabled = {
			modifier:diplomatic_reputation >= 1
			OR = {
				custom_tooltip = {
					text = nor_novgorod_none_exist_tt
					AND = {
						NOT = { country_exists = c:NOV }
						NOT = { country_exists = c:MOS }
						NOT = { country_exists = c:RUS }
					}
				}
				custom_tooltip = {
					text = nor_novgorod_border_and_strength_tt
					AND = {
						OR = {
							is_neighbor_of = c:NOV
							is_neighbor_of = c:MOS
							is_neighbor_of = c:RUS
						}
						OR = {
							AND = {
								country_exists = c:NOV
								OR = {
									has_mutual_scripted_relation = { target = c:NOV type = relation_type:alliance }
									root = { relative_strength = { target ?= c:NOV value >= 1.0 } }
								}
							}
							AND = {
								NOT = { country_exists = c:NOV }
								country_exists = c:MOS
								OR = {
									has_mutual_scripted_relation = { target = c:MOS type = relation_type:alliance }
									root = { relative_strength = { target ?= c:MOS value >= 1.0 } }
								}
							}
							AND = {
								NOT = { country_exists = c:NOV }
								NOT = { country_exists = c:MOS }
								country_exists = c:RUS
								OR = {
									has_mutual_scripted_relation = { target = c:RUS type = relation_type:alliance }
									root = { relative_strength = { target ?= c:RUS value >= 1.0 } }
								}
							}
						}
					}
				}
			}
		}

		duration = 0

		on_completion = {
			if = {
				limit = { country_exists = c:NOV }
				c:NOV = { save_scope_as = novgorod_target }
			}
			else_if = {
				limit = { country_exists = c:MOS }
				c:MOS = { save_scope_as = novgorod_target }
			}
			else_if = {
				limit = { country_exists = c:RUS }
				c:RUS = { save_scope_as = novgorod_target }
			}
			if = {
				limit = {
					OR = {
						country_exists = c:NOV
						country_exists = c:MOS
						country_exists = c:RUS
					}
				}
				if = {
					limit = { root = { relative_strength = { target = scope:novgorod_target value >= 1.0 } } }
					add_casus_belli = { target = scope:novgorod_target type = casus_belli:cb_conquer_province years = 25 }
					add_country_modifier = { modifier = novgorod_frontier_dominance years = 30 mode = add }
				}
				else = {
					add_country_modifier = { modifier = treaty_of_novgorod years = 30 mode = add }
				}
			}
			else = {
				add_country_modifier = { modifier = novgorod_frontier_dominance years = 30 mode = add }
			}
		}
	}
