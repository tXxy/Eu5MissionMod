		enabled = {
			is_during_bankruptcy = no
			OR = {
				# Military route: already conquered (or your subjects hold) all three locations
				AND = {
					location:visby = { is_owned_or_owned_by_subjects_or_below_of = root }
					location:slite = { is_owned_or_owned_by_subjects_or_below_of = root }
					location:hoborg = { is_owned_or_owned_by_subjects_or_below_of = root }
					army_tradition >= 40
					OR = {
						NOT = { country_exists = c:SWE }
						root = { relative_strength = { target ?= c:SWE value >= 0.90 } }
					}
				}
				# Diplomatic route: Sweden is bound to Denmark by union — demand it instead
				c:SWE ?= {
					OR = {
						in_union_with = root
						is_subject_of = root
					}
				}
			}
		}
		duration = 0
		on_completion = {
			if = {
				limit = {
					NOT = { owns = location:visby }
				}
				trigger_event_non_silently = denmark_mission_events.42
			}
			if = {
				limit = {
					owns = location:visby
				}
				location:visby = { add_core = root }
				location:slite = { add_core = root }
				location:hoborg = { add_core = root }
				add_army_tradition = 15
				add_prestige = 30
				add_gold = { value = monthly_income_trade_and_tax multiply = 15 }
				add_country_modifier = { modifier = gotland_seized years = 40 mode = add }
				location:visby = {
					add_location_modifier = { modifier = sack_of_visby years = 20 mode = add }
				}
			}
		}
