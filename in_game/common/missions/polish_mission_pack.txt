polish_mission_pack = {
	icon = generic_centralization
	chance = 3600
	repeatable = no
	player_playstyle = diplomatic

	visible = {
		game_has_missions_enabled = yes
		tag = "POL"
	}

	on_start = {
		set_variable = {
			name = mission_target_development
			value = 25
		}
		set_variable = {
			name = mission_target_manpower
			value = 15000
		}
	}

	on_completion = {
		add_prestige = 50
		add_legitimacy = 25
		remove_variable = mission_target_development
		remove_variable = mission_target_manpower
	}

	on_abort = {
		remove_variable = mission_target_development
		remove_variable = mission_target_manpower
	}

	# === POLISH UNIFICATION BRANCH ===
	
	mission_consolidate_polish_heartland = {
		icon = centralization_advance
		requires = { }
		
		enabled = {
			area:lesser_poland_area = {
				locations_owned_by_fraction = {
					target = root
					value >= 0.95
				}
			}
			area:greater_poland_area = {
				locations_owned_by_fraction = {
					target = root
					value >= 0.95
				}
			}
			stability >= 50
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = unified_polish_crown
				years = 10
				mode = add
			}
			add_legitimacy = 20
			add_gold = 100
			area:lesser_poland_area = {
				every_location_in_area = {
					change_control = control_mild_bonus
				}
			}
		}
	}

	mission_develop_krakow = {
		icon = cultural_acceptance_advance
		requires = { mission_consolidate_polish_heartland }
		
		enabled = {
			capital = {
				development >= 35
				total_building_levels >= 25
				has_building_with_at_least_one_level = marketplace
			}
			monthly_income >= var:mission_target_development
		}
		
		duration = 0

		on_completion = {
			add_gold = {
				value = monthly_income
				multiply = 8
			}
			capital = {
				add_development = 5
				add_location_modifier = {
					modifier = krakow_cultural_center
					years = 10
					mode = add
				}
			}
		}
	}

	mission_strengthen_szlachta = {

		enabled = {

				limit = { country_exists = c:TEU }
