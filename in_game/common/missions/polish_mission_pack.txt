# === 2. GRAIN TRADE ===
mission_establish_grain_trade = {
    icon = merchants_and_trade
    requires = { mission_develop_agricultural_heartland }
    
    enabled = {
				capital.market ?= {
						"merchant_power_in_market(root)" >= 35
				}
			
			
			# REQUIREMENT 3: ESTATE ALIGNMENT - Fixed version
			"estate_power(estate_type:burghers_estate)" >= 0.30

			# REQUIREMENT 4: PRODUCTION SURPLUS
				produced_in_country = {
						goods = goods:wheat
						value >= 30
				}

			# REQUIREMENT 4: PRODUCTION SURPLUS (New)
			# We must actually produce enough grain to trade.
			# This prevents completing the mission while having a famine.
			capital.market = {
					market_possible_goods_trade_surplus = {
							goods = goods:wheat
							country = root
							value >= 30 # Significant surplus required
					}
			}
    }
    
    duration = 0

    on_completion = {
        # Immediate Satisfaction Bonus
        add_estate_satisfaction = {
            type = estate_type:burghers_estate
            value = 0.05
        }
        
        # Triggering the narrative event
        trigger_event_non_silently = polish_mission_events.16
    }
		
}

			estate_satisfaction:nobles_estate >= 0.50
			estate_satisfaction:burghers_estate >= 0.50
			estate_satisfaction:peasants_estate >= 0.50
			
			# Government stability requirements
			stability >= 30
			# NOT = { has_country_modifier = noble_dominance_crisis }
			
			# Have some crown power to work with
			"estate_power(estate_type:crown_estate)" >= 0.25
    }
    
    duration = 730 # 2 Years to maintain the balance
    
    modifier_while_progressing = {
        stability_investment = 2
        global_monthly_control = 0.01
    }

    on_monthly = {
        random_list = {
            # Smart Triggers: Only fire complaints if satisfaction is low (< 60)
            15 = { 
                trigger_if = {
                    limit = { estate_satisfaction:nobles_estate < 60 }
                    trigger_event_silently = polish_mission_events.40 
                }
            } 
            10 = { 
                trigger_if = {
                    limit = { estate_satisfaction:burghers_estate < 60 }
                    trigger_event_silently = polish_mission_events.41 
                }
            } 
            10 = { 
                trigger_if = {
                    limit = { estate_satisfaction:peasants_estate < 50 }
                    trigger_event_silently = polish_mission_events.42 
                }
            }
            # High chance of nothing happening if you are managing well
            65 = { }
        }
    }

    on_completion = {
        trigger_event_silently = polish_mission_events.43
    }
		
}
