holland_naval_mission_pack = {
	icon = generic_conquer_province
	repeatable = no
	player_playstyle = military

	visible = {
		game_has_missions_enabled = yes
		tag = "HOL"
		num_of_ports > 2
	}

	enabled = {
		exists = capital
		navy_size > 15
	}
	
	chance = 3600
	
	on_start = {
		set_variable = {
			name = mission_target_navy_size
			value = {
				value = navy_size
				multiply = 1.5
				min = 30
			}
		}
		set_variable = {
			name = mission_target_heavy_ships
			value = {
				value = total_sub_unit_count:navy_heavy_ship
				add = 5
			}
		}
	}

	on_completion = {
		add_navy_tradition = 35
		add_prestige = 25
		remove_variable = mission_target_navy_size
		remove_variable = mission_target_heavy_ships
	}

	# === NAVAL DEVELOPMENT ===
	
	mission_modernize_shipyards = {
		icon = rudimentary_coastal_ship_repair
		requires = { }
		
		enabled = {
			total_effective_building_levels:shipyard >= 4
		}
		
		duration = 0

		on_completion = {
			every_owned_location = {
				limit = {
					has_building_with_at_least_one_level = shipyard
				}
				add_location_modifier = {
					modifier = holland_shipyard_efficiency
					years = 15
				}
			}
		}
	}

	mission_build_merchant_fleet = {
		icon = global_trade_routes_advance
		requires = { mission_modernize_shipyards }
		
		enabled = {
			total_sub_unit_count:navy_light_ship >= 20
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_merchant_fleet
				years = 15
			}
			add_navy_tradition = 15
		}
	}

	# === NAVAL SUPREMACY ===
	
	mission_protect_trade_routes = {
		icon = maritime_advance_age_2
		requires = { mission_build_merchant_fleet }
		
		enabled = {
			num_of_ports >= 8
			NOT = {
				any_neighbor_country = {
					navy_size > root.navy_size
				}
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_sea_lanes_secured
				years = 20
			}
			add_prestige = 25
		}
	}

	mission_build_battle_fleet = {
		icon = global_supply_limit_modifier_advance_5
		requires = { mission_modernize_shipyards }
		
		enabled = {
			navy_size >= var:mission_target_navy_size
			total_sub_unit_count:navy_heavy_ship >= var:mission_target_heavy_ships
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_grand_fleet
				years = 15
			}
			add_navy_tradition = 20
		}
	}

	# === COLONIAL EXPANSION ===
	
	mission_establish_colonial_company = {
		icon = colonial_heritage
		requires = { mission_protect_trade_routes }
		
		enabled = {
			num_colonies >= 2
			colonial_range > 200
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_colonial_company
				years = 25
			}
			add_colonist = 1
		}
	}

	mission_control_spice_islands = {
		icon = route_to_the_indies_advance
		requires = { mission_establish_colonial_company }
		
		enabled = {
			custom_tooltip = {
				text = control_indonesian_provinces_tt
				num_of_provinces_in_indonesia >= 3
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_spice_trade
				years = -1
			}
			add_gold = 400
		}
	}

	# === FINAL MISSION ===
	
	mission_naval_hegemony = {
		icon = global_trade_routes_advance
		requires = { mission_build_battle_fleet mission_control_spice_islands }
		final = yes
		
		enabled = {
			navy_size_percentage >= 0.8
			any_war = {
				OR = {
					attacker_leader = root
					defender_leader = root
				}
				naval_supermacy = yes
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_naval_hegemon
				years = -1
			}
			add_navy_tradition = 50
			add_prestige = 100
		}
	}
}
