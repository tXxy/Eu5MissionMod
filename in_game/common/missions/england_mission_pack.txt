	icon = crown_icon
	chance = 3600
	repeatable = no
	player_playstyle = diplomatic

			value = 25

		add_prestige = 100
		add_legitimacy = 75

		remove_variable = mission_target_wool_production

		remove_variable = mission_target_trade_income

		remove_variable = mission_target_wool_production
		remove_variable = mission_target_colonial_subjects
		remove_variable = mission_target_trade_income
		remove_variable = mission_target_parliament_support
