achaea_missions = {
    icon = generic_capital_economy
    repeatable = no
    chance = 3600   
    player_playstyle = diplomatic
    # ai_will_do = { value = 100 }   

    visible = {
        game_has_missions_enabled = yes
        OR = {
            has_or_had_tag = ACH
            has_or_had_tag = ATH
            has_or_had_tag = LAT_F

    }

    on_start = { 
        add_country_modifier = {
            modifier = ach_latin_legacy
            days = -1
            mode = add_and_extend

    }

    on_abort = {
        remove_country_modifier = ach_latin_legacy
    }

    on_completion = {
        add_prestige = 50
        add_legitimacy = 25
