holland_trade_mission_pack = {
	icon = generic_capital_economy
	repeatable = no
	player_playstyle = administrative

	visible = {
		game_has_missions_enabled = yes
		tag = "HOL"
	}

	enabled = {
		exists = capital
		num_of_ports > 0
	}
	
	chance = 3600

	on_start = {
		set_variable = {
			name = mission_target_trade_income
			value = {
				value = monthly_trade_income
				multiply = 2
				min = 5
			}
		}
		set_variable = {
			name = mission_target_development
			value = {
				value = capital.development
				add = 15
			}
		}
	}

	on_completion = {
		add_prestige = 30
		remove_variable = mission_target_trade_income
		remove_variable = mission_target_development
	}

	on_abort = {
		remove_variable = mission_target_trade_income
		remove_variable = mission_target_development
	}

	# === EARLY GAME: LOCAL CONSOLIDATION ===
	
	mission_consolidate_local_trade = {
		icon = global_trade_routes_advance
		requires = { }
		
		enabled = {
			monthly_trade_income >= var:mission_target_trade_income
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_local_markets_modifier
				years = 20
			}
			add_gold = 50
		}
	}

	mission_develop_amsterdam = {
		icon = merchants_and_trade
		requires = { mission_consolidate_local_trade }
		
		enabled = {
			capital = {
				development >= var:mission_target_development
				total_effective_building_levels:marketplace >= 2
			}
		}
		
		duration = 0

		on_completion = {
			capital = {
				add_location_modifier = {
					modifier = holland_amsterdam_trade_hub
					years = 15
				}
			}
			add_gold = 100
		}
	}

	# === MID GAME: REGIONAL DOMINANCE ===
	
	mission_channel_trade_dominance = {
		icon = diplomatic_influence
		requires = { mission_develop_amsterdam }
		
		enabled = {
			capital.market = {
				has_merchant = root
				trade_power_share = {
					country = root
					value >= 0.25
				}
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_channel_trade_modifier
				years = 15
			}
		}
	}

	mission_estate_cooperation = {
		icon = court_accounting
		requires = { mission_develop_amsterdam }
		
		enabled = {
			any_estate = {
				estate_type = estate_type:burghers_estate
				loyalty >= 50
				influence >= 0.4
			}
		}
		
		duration = 0

		on_completion = {
			capital = {
				add_location_modifier = {
					modifier = holland_guild_production_boost
					years = 15
				}
			}
			add_estate_loyalty = {
				type = estate_type:burghers_estate
				value = 10
			}
		}
	}

	# === LATE GAME: GLOBAL EXPANSION ===
	
	mission_establish_voc = {
		icon = colonial_heritage
		requires = { mission_channel_trade_dominance mission_estate_cooperation }
		
		enabled = {
			num_of_ports >= 6
			any_estate = {
				estate_type = estate_type:burghers_estate
				influence >= 0.6
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_voc_charter
				years = 25
			}
			add_gold = 300
		}
	}

	mission_banking_innovations = {
		icon = correct_box_advance_renaissance
		requires = { mission_channel_trade_dominance }
		
		enabled = {
			has_embraced_institution = institution:banking
			research_speed_modifier >= 0.1
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_financial_center_modifier
				years = 25
			}
			add_gold = 200
		}
	}

	# === FINAL MISSION ===
	
	mission_dutch_trade_hegemony = {
		icon = wealth_renaissance
		requires = { mission_establish_voc mission_banking_innovations }
		final = yes
		
		enabled = {
			monthly_trade_income >= 50
			capital = {
				development >= 40
			}
		}
		
		duration = 0

		on_completion = {
			add_country_modifier = {
				modifier = holland_trade_mastery_modifier
				years = -1
			}
			add_gold = 500
			add_prestige = 75
			custom_tooltip = holland_trade_hegemony_tt
		}
	}
}
