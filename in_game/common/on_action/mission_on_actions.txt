on_ending_war = {
    effect = {
        # -----------------------------------------------------------------
        # Polish Mission: HRE Rivalry Outcome
        # -----------------------------------------------------------------
        
        # SCENARIO 1: Poland (Root) WINS
        if = {
            limit = {
                # 1. Did the winner start this specific scripted war?
                scope:winner = { 
                    has_variable = hre_war_declared 
                }
                
                # 2. Was the loser the specific target we marked?
                scope:loser = { 
                    has_variable = is_polish_mission_target 
                }
                
                # 3. Check for the specific Humiliate CB used in the event
                scope:war = {
                    casus_belli = casus_belli:cb_humiliate
                }
            }
            
            # Effects for the Winner (Poland)
            scope:winner = {
                remove_variable = hre_war_declared
                remove_variable = waiting_for_hre_war_outcome
                
                # Trigger the Victory Event
                trigger_event_non_silently = polish_mission_events.81
            }
            
            # Cleanup for the Loser
            scope:loser = {
                remove_variable = is_polish_mission_target
            }
        }
        
        # SCENARIO 2: Poland (Root) LOSES
        else_if = {
            limit = {
                # 1. Did the LOSER start this specific scripted war?
                scope:loser = { 
                    has_variable = hre_war_declared 
                }
                
                # 2. Was the WINNER the specific target?
                scope:winner = { 
                    has_variable = is_polish_mission_target 
                }
                
                # 3. Check for the CB
                scope:war = {
                    casus_belli = casus_belli:cb_humiliate
                }
            }
            
            # Effects for the Loser (Poland)
            scope:loser = {
                remove_variable = hre_war_declared
                remove_variable = waiting_for_hre_war_outcome
                
                # Trigger the Defeat Event
                trigger_event_non_silently = polish_mission_events.83
            }
            
            # Cleanup for the Winner
            scope:winner = {
                remove_variable = is_polish_mission_target
            }
        }
    }
}
