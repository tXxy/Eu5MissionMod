# ==================== EVENT 1: SECURING THE SWEDISH CROWN ====================
sweden_mission_events.1 = {
	type = country_event
	category = mission_event
	title = sweden_mission_events.1.t
	desc = sweden_mission_events.1.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:nobles_estate 
			background = estate_type:crown_estate

		name = sweden_mission_events.1.a
		custom_tooltip = sweden_mission_events.1.a.tt

			type = estate_type:nobles_estate
			value = estate_satisfaction_severe_bonus
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_left
		}
	}

	option = {
		name = sweden_mission_events.1.b
		custom_tooltip = sweden_mission_events.1.b.tt
		
		# Focus on Centralization
		add_legitimacy = 20
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
	}
}

			add_cultural_tradition = cultural_tradition_mild_bonus

				limit = { owner ?= root }
				order_by = development
				max = 4
				add_location_modifier = {
					modifier = integrated_swedish_heartland
					years = 20
					mode = add
				}
			}
		}
	}

		10 = regular

	}

	immediate = {
		event_illustration_poptype_effect = { 
			foreground = pop_type:soldiers 
			background = pop_type:soldiers 
		}
	}

	option = {
		name = sweden_mission_events.5.a
		custom_tooltip = sweden_mission_events.5.a.tt
		
		# Create Elite Commander
		create_character = {
			first_name = Gustav
			last_name = Horn
			adm = 85
			dip = 65
			mil = 95
			estate = estate_type:nobles_estate
			culture = culture:swedish
			religion ?= religion:lutheran
			add_trait = trait:born_to_the_saddle
			age = 32
			save_scope_as = carolean_commander
			create_in_limbo = yes
		}
		
		if = {
			limit = { exists = scope:carolean_commander }
			scope:carolean_commander = {
				move_country = root
			}
		}
	}

	option = {
		name = sweden_mission_events.5.b
		custom_tooltip = sweden_mission_events.5.b.tt

		add_manpower = {
			value = max_manpower
			multiply = 0.25
		}

		add_army_tradition = 25
	}
}

	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:nobles_estate 
			background = estate_type:nobles_estate 
		}
		location:kalmar ?= {
			save_scope_as = kalmar_capital
		}
	}

	option = {
		name = sweden_mission_events.8.a
		custom_tooltip = sweden_mission_events.8.a.tt
		
		# Union Focus
		scope:kalmar_capital = {
			change_development = 3
			add_location_modifier = {
				modifier = union_capital_integrated
				years = 20
				mode = add
			}
		}
		every_subject = {
			limit = { culture ?= { has_culture_group = culture_group:scandinavian_group } }
			add_liberty_desire = -20
			add_trust = {
				modifier = trust_mission_trusted_ties
				target = root
			}
		}
	}

	option = {
		name = sweden_mission_events.8.b
		custom_tooltip = sweden_mission_events.8.b.tt
		
		# Resource Extraction Focus
		region:scandinavian_region = {
			ordered_location_in_region = {
				limit = { owner ?= root raw_material = goods:lumber }
				order_by = development
				max = 6
				change_prosperity = prosperity_weak_bonus
				change_max_raw_material_workers = 1
			}
		}
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 5
		}
	}
}
