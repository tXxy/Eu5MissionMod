polish_mission_events.8 = {
    type = country_event
    category = mission_event
    title = polish_mission_events.8.t
    desc = polish_mission_events.8.d
    
    illustration_tags = {
        10 = empire
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { 
            foreground = estate_type:nobles_estate 
            background = estate_type:nobles_estate 
        }
        
        # CRITICAL FIX: Robust scope saving
        # We scope to the IO, then grab the leader from there
        international_organization:hre = {
            leader_country = {
                save_scope_as = target_country
            }
        }
        
        # Set tracking variables
        set_variable = {
            name = hre_challenge_initiated
            value = yes
        }
        # Note: Saving complex objects (scopes) in variables can be unstable in some PDX versions.
        # It is safer to rely on 'save_scope_as' within the event chain, 
        # or use a flag on the target country itself.
        scope:target_country = {
            set_variable = {
                name = is_polish_mission_target
                value = yes
                years = 5
            }
        }
    }

    # Option A: War
    option = {
        name = polish_mission_events.8.a
        
        # 1. Add Opinion (Now safe because scope is guaranteed from immediate)
        scope:target_country = {
            add_opinion = { 
                target = root 
                modifier = opinion_polish_aggressions 
            }
        }

        # 2. CRITICAL: Add the CB FIRST, then declare war
        if = {
            limit = {
                exists = scope:target_country
            }
            add_casus_belli = {
                target = scope:target_country
                type = casus_belli:cb_humiliate # Ensure this CB type exists in your common/casus_belli folder
                years = 5 # Gives the CB a duration just in case the war command fails
            }
            declare_war_with_cb = {
                target = scope:target_country
                type = casus_belli:cb_humiliate
            }
        }

        add_country_modifier = {
            modifier = imperial_challenger
            years = 5
            mode = add
        }
        
        # War Tracking
        set_variable = {
            name = hre_war_declared
            value = yes
        }
        set_variable = {
            name = waiting_for_hre_war_outcome
            value = yes
        }
    }

    # Option B: Diplomacy / Subversion
    option = {
        name = polish_mission_events.8.b
        
        add_country_modifier = {
            modifier = diplomatic_maneuvering
            years = 8
            mode = add
        }
        
        # Logic: Find HRE provinces in Germany that are NOT ours, and give claims
        random_location = {
            limit = {
                region = region:germany_region
                owner = {
                    is_member_of_international_organization = international_organization:hre
                    NOT = { this = root }
                    NOT = { is_subject_of = root }
                }
            }
            max = 3
            add_claim = root
        }
        
        add_prestige = 25
        
        # Improve relations with HRE members who dislike the emperor
        every_country = {
            limit = {
                is_member_of_international_organization = international_organization:hre
                NOT = { this = root }
                # Only affect those who actually dislike the Emperor
                opinion = {
                    target = scope:target_country
                    value < 0
                }
            }
            add_opinion = {
                target = root
                modifier = opinion_common_rival # Ensure this modifier is defined in common/opinion_modifiers
            }
        }
        
        custom_tooltip = {
            text = polish_mission_events.8.b.tooltip_support
        }
    }
}

polish_mission_events.81 = {
    type = country_event
    category = mission_event
    title = polish_mission_events.81.t
    desc = polish_mission_events.81.d
    
    fire_only_once = yes 
    
    illustration_tags = {
        10 = victory
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = { 
            foreground = estate_type:nobles_estate 
            background = estate_type:nobles_estate 
        }
        
        # Find the Emperor using the flag
        every_country = { 
            limit = { has_variable = is_polish_mission_target }
            save_scope_as = current_emperor
            remove_variable = is_polish_mission_target
        }
    }

    option = {
        name = polish_mission_events.81.a
        
        add_prestige = 50
        add_power_projection = 25
        
        add_country_modifier = {
            modifier = hre_counterweight_permanent
            years = -1
            mode = add
        }
        
        # Weaken the HRE emperor
        # (I kept ONLY this one block. The duplicate is deleted.)
        scope:current_emperor = {
            add_prestige = -25
            add_legitimacy = -20
            
            if = {
                limit = {
                    prestige < 25
                    legitimacy < 40
                }
                random_list = {
                    30 = {
                        trigger_event_silently = {
                            id = hre.801
                        }
                        custom_tooltip = polish_mission_events.81.a.tt1
                    }
                    70 = { } 
                }
            }
        }
        
        # FIX: Claims on Border Territories
        every_neighbor_country = {
            limit = {
                is_member_of_international_organization = international_organization:hre
            }
            # Use ordered instead of random to ensure we get valid border tiles
            ordered_owned_location = {
                limit = {
                    # CRITICAL: Only claim locations that actually touch us
                    is_neighbor_of = root
                    NOT = { has_claim = root }
                    NOT = { is_core = root }
                }
                # Prioritize high pop or development so we claim good land
                order_by = population
                max = 2
                add_claim = root
            }
        }
    }
}

            # We do NOT remove the variable here yet,

            # If this is the end of the chain, you can uncomment the line below:
            # remove_variable = is_polish_mission_target
        }
    }

    option = {
        name = polish_mission_events.82.a
        # Our diplomatic efforts bear fruit
        
        add_diplomatic_reputation = 2
        add_improve_relations_modifier = 20
        
        # Chance to gain allies within HRE
        every_country = {
            limit = {
                is_member_of_international_organization = international_organization:hre

                is_rival_of = scope:current_emperor
                NOT = { has_mutual_scripted_relation = { type = relation_type:alliance target = root } }

            random_list = {
                40 = {
                    create_alliance = {
                        target = prev
                        from = root
                    }
                    custom_tooltip = polish_mission_events.82.a.tt1
                }
                60 = { } 
            }
        }
        
        add_country_modifier = {
            modifier = hre_economic_influence
            years = 10
            mode = add
        }
    }

    option = {
        name = polish_mission_events.82.b
        # Shift to more aggressive stance
        
        remove_country_modifier = diplomatic_maneuvering
        
        add_country_modifier = {
            modifier = preparing_hre_confrontation
            years = 3
            mode = add
        }
        
        # FIX: Use scope:current_emperor logic
        if = {
            limit = {
                scope:current_emperor = {
                    prestige < 30
                    has_royal_marriage_with = root
                    same_dynasty_as = root
                }
            }
            add_casus_belli = {
                target = scope:current_emperor
                type = casus_belli:cb_restore_personal_union
                months = 60
            }
        }
    }

polish_mission_events.15 = {
    type = country_event
    category = mission_event
    title = polish_mission_events.15.t
    desc = polish_mission_events.15.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = production
        10 = exterior
    }

    immediate = {
        # Defensive coding: Ensure estate exists before querying art
        if = {
            limit = { has_estate = estate_nobles }
            event_illustration_estate_effect = { 
                foreground = estate_type:nobles_estate 
                background = estate_type:nobles_estate 
            }
        }
        # Fallback illustration if nobles are gone (optional but recommended)
        else = {
             event_illustration = { texture = "gfx/event_pictures/generic_fields.dds" }
        }
    }

    option = {
        name = polish_mission_events.15.a
                
        add_estate_power = {
            type = estate_type:nobles_estate
            value = 0.10
        }
        
        every_owned_location = {
            limit = { raw_material = goods:wheat }
						#not needed the other buffs are already enough
            # add_location_modifier = {
            #     modifier = wheat_production_efficiency
            #     years = 15
            #     mode = add
            # }
            # REFACTOR 4: Explicit value block for satisfaction
            every_pop = {
                limit = { pop_type = pop_type:peasants }
                add_pop_satisfaction = {
                    value = pop_satisfaction_severe_penalty
                }
            }
        }


				#Population increase requires script math for percentage
        # The original '0.05' would be cast to 0. We use a math block.
        every_owned_location = {
            limit = { raw_material = goods:wheat }
            every_pop = {
                limit = { pop_type = pop_type:peasants }
                # Intention: +5% population growth
                add_pop_size = {
                    value = pop_size # Fetch current integer size
                    multiply = 0.05  # Calculate 5%
                }
            }
        }
    }

    option = {
        name = polish_mission_events.15.b
        
        add_country_modifier = {
            modifier = rural_investment
            years = 15
            mode = add
        }
        
        add_estate_satisfaction = {
            type = estate_type:peasants_estate
            value = estate_satisfaction_mild_bonus
        }
				add_estate_satisfaction = {
            type = estate_type:nobles_estate
            value = estate_satisfaction_severe_penalty
        }        
    }
}

polish_mission_events.16 = {
	type = country_event
	category = mission_event
	title = polish_mission_events.16.t
	desc = polish_mission_events.16.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = trade
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:burghers_estate 
			background = estate_type:burghers_estate 
		}
		# Save scopes for more dynamic event
		capital = {
			save_scope_as = trade_hub_location
		}
		goods:wheat = {
			save_scope_as = target_good
		}
	}

	option = {
		name = polish_mission_events.16.a
		# Focus on immediate profit through exports
		
		# Create temporary demand like in flavor_hol.33
		 capital.market ?= {
				add_goods_supply = {
						goods = goods:wheat
						amount = market_goods_severe_gain
				}
			}
		
		# Immediate gold boost based on trade infrastructure
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 8
		}
		
		# Boost burghers but strain peasants (like in flavor_sco.18)
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_mild_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:peasants_estate
			value = estate_satisfaction_weak_penalty
		}
		
		# Set variable for potential follow-up event like flavor_hol.34
		set_variable = {
			name = polish_grain_export_initiated
			value = yes
		}
	}

	option = {
		name = polish_mission_events.16.b
		# Focus on trade infrastructure development
		# Invest in trade buildings like official marketplace mission
		ordered_owned_rural_location = {
			limit = {
				raw_material = goods:wheat
				has_building_with_at_least_one_level = windmill
			}
			order_by = development
			max = 3
			change_development = 5
			change_prosperity = prosperity_weak_bonus
			add_location_modifier = {
				modifier = grain_trade_infrastructure_boost
				years = 5
				mode = add
			}
		}
		
		# Strong burghers satisfaction boost
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_severe_bonus
		}
	}

	option = {
		name = polish_mission_events.16.c
		# Balanced approach - domestic market development
		# Increasing Market Access in the capital region
			capital.market = {
					add_merchant_power = {
							country = root
							key = market_dominance_key
							years = 15
							power = 25 # Massive boost to trade power
					}
			}
		
		# Balanced estate satisfaction
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_weak_bonus
		}
		
		# Small trade income modifier
		add_country_modifier = {
			modifier = balanced_grain_economy
			years = 8
			mode = add
		}
	}
}

        }
        
        add_estate_satisfaction = {
            type = estate_type:nobles_estate
            value = estate_satisfaction_mild_bonus
        }
    }

    # Option B: The Quarter Army System (Professionalism Focus)
    option = {
        name = polish_mission_events.21.b
        
        change_gold_effect = { 
            scale = -15
        }
        
        # Manpower and professionalism boost
        add_manpower = {
					value = max_manpower
					multiply = 0.15
				}
        
        add_army_professionalism = 0.10
        
        # Permanent reform modifier
        add_country_modifier = {
            modifier = plc_standing_army_reforms
            years = -1  # Permanent
        }
        
        change_societal_value = {
            type = offensive_vs_defensive  
            value = societal_value_move_to_left
        }
    }

    # Option C: Noble Levy Modernization (Compromise)
    option = {
        name = polish_mission_events.21.c
        
        add_army_tradition = army_tradition_mild_bonus
        
        # Balanced approach - some benefits with less cost
        add_country_modifier = {
            modifier = polish_noble_levy_reforms
            years = 15
        }
        
        # Less estate impact
        add_estate_satisfaction = {
            type = estate_type:nobles_estate
            value = estate_satisfaction_weak_bonus
        }
        
        add_prestige = prestige_mild_bonus
