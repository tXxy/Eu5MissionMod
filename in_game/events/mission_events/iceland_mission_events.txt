# ==================== EVENT 1: ASSERT ICELANDIC LAW ====================
iceland_mission_events.1 = {
	type = country_event
	category = mission_event
	title = iceland_mission_events.1.t
	desc = iceland_mission_events.1.d
	fire_only_once = yes
	is_triggered_only = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:nobles_estate
		}
	}

	option = {
		name = iceland_mission_events.1.a

		add_legitimacy = 20
		add_stability = 10
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
	}

	option = {
		name = iceland_mission_events.1.b

		add_gold = 200
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
	}
}

# ==================== EVENT 11: ÖRÆFAJÖKULL ERUPTS (automatic, unavoidable) ====================
# Fires on its own around 1362, independent of mission progress — mirrors how
# irish_mission_events.10 (famine_of_1740) and sack_of_visby work: not
# is_triggered_only, just a trigger window + mean_time_to_happen.
iceland_mission_events.11 = {
	type = country_event
	category = mission_event
	title = iceland_mission_events.11.t
	desc = iceland_mission_events.11.d

	fire_only_once = yes

	trigger = {
		tag = "ICE"
		year >= 1362
		year <= 1364
	}

	mean_time_to_happen = {
		months = 6
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:peasants_estate
			background = estate_type:crown_estate
		}
	}

	option = {
		name = iceland_mission_events.11.a

		add_country_modifier = {
			modifier = oraefajokull_eruption
			years = 3
			mode = add
		}
		add_stability = -5
		add_prestige = -10
	}
}

# ==================== EVENT 13: A PLEA TO THE NORWEGIAN CROWN ====================
# Norway (if it still holds any of Greenland) formally cedes it to Iceland.
# Only reachable once the steep opinion/reputation bar in the mission is met,
# so a single guaranteed option is appropriate here — the hard part was
# earning the favor, not the asking.
iceland_mission_events.13 = {
	type = country_event
	category = mission_event
	title = iceland_mission_events.13.t
	desc = iceland_mission_events.13.d
	fire_only_once = yes
	is_triggered_only = yes

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:burghers_estate
		}
	}

	option = {
		name = iceland_mission_events.13.a

		if = {
			limit = {
				country_exists = c:NOR
			}
			add_gold = -200
			c:NOR = {
				add_gold = 200
			}
		}
		area:greenland_area = {
			every_location_in_area = {
				limit = {
					is_owned_or_owned_by_subjects_or_below_of = c:NOR
				}
				remove_core = c:NOR
				change_location_owner = root
				add_core = root
			}
		}
		add_prestige = 25
		add_legitimacy = 10
	}
}
