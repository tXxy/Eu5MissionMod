namespace = byz_mission_events

# ====================================================================
# EVENT 1: "The Theodosian Walls Restored" (Secure Constantinople)
# ====================================================================
byz_mission_events.1 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.1.t
    desc = byz_mission_events.1.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = defensive
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:nobles_estate
            background = estate_type:peasants_estate
        }
        capital = {
            save_scope_as = constantinople_location
        }
        # Save current ruler for dynamic effects
        ruler_or_regent ?= {
            save_scope_as = current_ruler
        }
    }

    # Option A: The Grand Restoration (Historical & Prestige)
    option = {
        name = byz_mission_events.1.a
        historical_option = yes
        
        add_prestige = prestige_mild_bonus
        
        custom_tooltip = {
            text = byz_mission_events.1.a.tt
            hidden_effect = {
                capital = {
                    # Additional defensive building if we have the tech
                    if = {
                        limit = {
                            has_advance = bastion_advance
                        }
                        add_building_level = {
                            building = building_type:bastion
                            levels = 1
                        }
                    }
                    else_if = {
                        limit = {
                            has_advance = castle_advance
                        }
                        add_building_level = {
                            building = building_type:castle
                            levels = 1
                        }
                    }
                }
            }
        }
        
        # Dynamic stability based on ruler's military skill
        add_stability = {
            value = stability_mild_bonus
            add = {
                value = scope:current_ruler.mil
                divide = 10
                min = 0
                max = 20
            }
        }
        
        # Set flag for future events
        set_variable = {
            name = walls_restored_grandly
            value = 1
        }
    }

    # Option B: Practical Repairs (Cheaper, Less Impressive)
    option = {
        name = byz_mission_events.1.b
        
        custom_tooltip = {
            text = byz_mission_events.1.b.tt
            hidden_effect = {
                capital = {
                    add_location_modifier = {
                        modifier = practical_fortifications
                        years = 15
                        mode = add
                    }
                }
            }
        }
        
        # Save gold but lose some prestige
        add_prestige = prestige_weak_penalty
        
        # Slight manpower boost from using local labor
        add_manpower = {
            value = max_manpower
            multiply = 0.05
        }
        
        set_variable = {
            name = walls_restored_practically
            value = 1
        }
    }

    # Option C: Reinforce the Sea Walls (Naval Focus)
    option = {
        name = byz_mission_events.1.c
        
        trigger = {
            navy_size >= 10
            capital = {
                has_building_with_at_least_one_level = dock
            }
        }
        
        add_navy_tradition = navy_tradition_mild_bonus
        
        custom_tooltip = {
            text = byz_mission_events.1.c.tt
            hidden_effect = {
                capital = {
                    add_location_modifier = {
                        modifier = reinforced_sea_walls
                        years = 20
                        mode = add
                    }
                }
            }
        }
        
        # Boost to coastal defense
        change_societal_value = {
            type = land_vs_naval
            value = societal_value_minor_move_to_right
        }
        
        set_variable = {
            name = sea_walls_enhanced
            value = 1
        }
    }
}

# EVENT 2: "The Hexamilion Reconstructed" (Greek Defense Line)
# ====================================================================
byz_mission_events.2 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.2.t
    desc = byz_mission_events.2.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = construction
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:peasants_estate
            background = estate_type:nobles_estate
        }
        
        # Save key locations
        if = {
            limit = {
                exists = location:corinth
            }
            location:corinth = {
                save_scope_as = corinth_location
            }
        }
        
        area:morea_area = {
            ordered_location_in_area = {
                limit = {
                    owner ?= root
                    has_river = yes
                }
                order_by = development
                max = 1
                save_scope_as = primary_quarry
            }
        }
    }

    # Option A: The Historical Hexamilion (Defensive Focus)
    option = {
        name = byz_mission_events.2.a
        historical_option = yes
        
        add_country_modifier = {
            modifier = hexamilion_complete
            years = 25
            mode = add
        }
        
        custom_tooltip = {
            text = byz_mission_events.2.a.tt
            hidden_effect = {
                # Build the wall at Corinth
                scope:corinth_location = {
                    if = {
                        limit = {
                            NOT = { has_building = building_type:hexamilion_wall }
                        }
                        add_building_level = {
                            building = building_type:hexamilion_wall
                            levels = 1
                        }
                    }
                    else = {
                        change_building_level_in_location = {
                            building = building_type:hexamilion_wall
                            value = 1
                        }
                    }
                }
            }
        }
        
        # Strong defensive bonus but expensive
        change_gold_effect = { scale = -5 }
        
        add_prestige = prestige_mild_bonus
    }

    # Option B: Fortified Isthmus (Network of Forts)
    option = {
        name = byz_mission_events.2.b
        
        add_country_modifier = {
            modifier = fortified_isthmus
            years = 20
            mode = add
        }
        
        custom_tooltip = {
            text = byz_mission_events.2.b.tt
            hidden_effect = {
                # Build multiple forts along the isthmus
                area:morea_area = {
                    every_location_in_area = {
                        limit = {
                            owner ?= root
                            is_coastal = yes
                        }
                        max = 3
                        if = {
                            limit = {
                                has_advance = bastion_advance
                                NOT = { has_building = building_type:bastion }
                            }
                            add_building_level = {
                                building = building_type:bastion
                                levels = 1
                            }
                        }
                        else_if = {
                            limit = {
                                has_advance = castle_advance
                                NOT = { has_building = building_type:castle }
                            }
                            add_building_level = {
                                building = building_type:castle
                                levels = 1
                            }
                        }
                    }
                }
            }
        }
        
        change_gold_effect = { scale = -8 }
        
        # Higher maintenance but better defense
        add_manpower = {
            value = max_manpower
            multiply = 0.1
        }
    }

    # Option C: Watchtower Network (Cheaper, Detection Focus)
    option = {
        name = byz_mission_events.2.c
        
        add_country_modifier = {
            modifier = peloponnese_watchtowers
            years = 15
            mode = add
        }
        
        custom_tooltip = {
            text = byz_mission_events.2.c.tt
            hidden_effect = {
                # Build watchtowers
                area:morea_area = {
                    ordered_location_in_area = {
                        limit = {
                            owner ?= root
                            topography = hills
                        }
                        order_by = development
                        max = 5
                        add_location_modifier = {
                            modifier = watchtower_network
                            years = 10
                            mode = add
                        }
                    }
                }
            }
        }
        
        change_gold_effect = { scale = -2 }
        
        # Boost to army tradition from improved scouting
        add_army_tradition = army_tradition_weak_bonus
    }
}

# EVENT 3: "Nicaea Liberated" (First Major Anatolian Victory)
# ====================================================================
byz_mission_events.3 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.3.t
    desc = byz_mission_events.3.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = religious
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:clergy_estate
            background = estate_type:peasants_estate
        }
        
        location:iznik = {
            save_scope_as = nicaea_location
            province = {
                save_scope_as = nicaea_province
            }
        }
        
        # Check for Turkish presence
        if = {
            limit = {
                country_exists = c:TUR
            }
            c:TUR = {
                save_scope_as = turkish_threat
            }
        }
        
        # Save religious unity in the region
        area:marmara_area = {
            save_scope_as = marmara_area_scope
        }
    }

    # Option A: Re-establish the Patriarchate (Religious Focus)
    option = {
        name = byz_mission_events.3.a
        historical_option = yes
        
        add_legitimacy = 30
        add_religious_influence = religious_influence_severe_bonus
        
        custom_tooltip = {
            text = byz_mission_events.3.a.tt
            hidden_effect = {
                scope:nicaea_location = {
                    # Convert population
                    every_pop = {
                        limit = {
                            owner = root
                            religion != religion:orthodox
                        }
                        change_pop_religion = religion:orthodox
                    }
                    
                    # Build cathedral if not present
                    if = {
                        limit = {
                            NOT = { has_building = building_type:cathedral }
                        }
                        add_building_level = {
                            building = building_type:cathedral
                            levels = 1
                        }
                    }
                    
                    add_location_modifier = {
                        modifier = nicaea_patriarchate_restored
                        years = 20
                        mode = add
                    }
                }
            }
        }
        
        # Trigger follow-up event about religious councils
        set_variable = {
            name = nicaea_council_planned
            value = 1
        }
    }

    # Option B: Fortify the City (Military Focus)
    option = {
        name = byz_mission_events.3.b
        
        add_army_tradition = army_tradition_mild_bonus
        
        custom_tooltip = {
            text = byz_mission_events.3.b.tt
            hidden_effect = {
                scope:nicaea_location = {
                    # Strong fortifications
                    if = {
                        limit = {
                            has_advance = castle_advance
                        }
                        add_building_level = {
                            building = building_type:castle
                            levels = 1
                        }
                    }
                    
                    add_location_modifier = {
                        modifier = nicaea_fortified_bridgehead
                        years = 15
                        mode = add
                    }
                }
                
                # Claims on surrounding Turkish territory
                if = {
                    limit = {
                        exists = scope:turkish_threat
                    }
                    scope:turkish_threat = {
                        area:marmara_area = {
                            every_location_in_area = {
                                limit = {
                                    owner ?= scope:turkish_threat
                                }
                                max = 4
                                add_claim = root
                            }
                        }
                    }
                }
            }
        }
        
        # Slight penalty with clergy for militaristic approach
        add_estate_satisfaction = {
            type = estate_type:clergy_estate
            value = estate_satisfaction_weak_penalty
        }
    }

    # Option C: Re-settle with Greeks (Cultural Focus)
    option = {
        name = byz_mission_events.3.c
        
        culture = {
            add_cultural_influence = cultural_influence_mild_bonus
        }
        
        custom_tooltip = {
            text = byz_mission_events.3.c.tt
            hidden_effect = {
                # Migration of Greek population
                add_migration = {
                    owner = this
                    from = capital.province_definition
                    to = scope:nicaea_province
                    culture = culture:greek_culture
                    amount = 0.08
                    months = 60
                    type = pop_type:burghers
                }
                
                scope:nicaea_location = {
                    # Cultural buildings
                    if = {
                        limit = {
                            NOT = { has_building = building_type:library }
                        }
                        add_building_level = {
                            building = building_type:library
                            levels = 1
                        }
                    }
                    
                    add_location_modifier = {
                        modifier = greek_resettlement_program
                        years = 12
                        mode = add
                    }
                }
            }
        }
        
        # Boost to development
        scope:nicaea_location = {
            change_development = development_mild_bonus
        }
    }
}

# EVENT 4: "The Ottoman Threat Ended" (Breaking Ottoman Power)
# ====================================================================
byz_mission_events.4 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.4.t
    desc = byz_mission_events.4.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = victory
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:nobles_estate
            background = estate_type:soldiers
        }
        
        # Check how the Ottomans were defeated
        if = {
            limit = {
                NOT = { country_exists = c:TUR }
            }
            set_variable = {
                name = ottomans_destroyed
                value = 1
            }
        }
        else_if = {
            limit = {
                c:TUR = {
                    is_subject_of = root
                }
            }
            set_variable = {
                name = ottomans_vassalized
                value = 1
            }
            c:TUR = {
                save_scope_as = turkish_vassal
            }
        }
        
        location:bursa = {
            save_scope_as = bursa_location
        }
        
        # Save current military strength
        save_scope_as = byzantium_country
    }

    # Option A: The Final Campaign (If Ottomans destroyed)
    option = {
        name = byz_mission_events.4.a
        trigger = {
            has_variable = ottomans_destroyed
        }
        
        add_prestige = prestige_severe_bonus
        add_army_tradition = army_tradition_severe_bonus
        
        custom_tooltip = {
            text = byz_mission_events.4.a.tt
            hidden_effect = {
                # Permanent claims on former Ottoman core lands
                region:anatolia_region = {
                    every_location_in_region = {
                        limit = {
                            has_owner = yes
                            owner = {
                                culture = culture:turkish_culture
                            }
                        }
                        max = 8
                        add_permanent_claim = root
                    }
                }
                
                # Development boost to Bursa as new Byzantine center
                scope:bursa_location = {
                    change_development = development_severe_bonus
                    change_prosperity = prosperity_mild_bonus
                }
            }
        }
        
        # Massive legitimacy boost
        add_legitimacy = 50
        
        # Trigger follow-up event about dividing the spoils
        set_variable = {
            name = ottoman_spoils_available
            value = 1
        }
    }

    # Option B: The Humiliating Peace (If Ottomans vassalized)
    option = {
        name = byz_mission_events.4.b
        trigger = {
            has_variable = ottomans_vassalized
        }
        
        add_prestige = prestige_mild_bonus
        
        custom_tooltip = {
            text = byz_mission_events.4.b.tt
            hidden_effect = {
                # Extract tribute and concessions
                scope:turkish_vassal = {
                    add_subject_tribute_modifier = {
                        years = 10
                        value = 0.25
                    }
                    
                    # Force religion change if possible
                    if = {
                        limit = {
                            religion != religion:orthodox
                        }
                        add_opinion = {
                            target = root
                            modifier = opinion_forced_conversion
                        }
                    }
                    
                    # Take strategic locations
                    capital = {
                        province = {
                            every_location_in_province = {
                                add_claim = root
                            }
                        }
                    }
                }
            }
        }
        
        # Boost to army professionalism from managing vassal
        add_army_professionalism = 0.05
        
        # Set up event chain about managing the Turkish vassal
        set_variable = {
            name = managing_turkish_vassal
            value = 1
        }
    }

    # Option C: Divide and Rule (Diplomatic Approach)
    option = {
        name = byz_mission_events.4.c
        
        add_diplomatic_reputation = 1
        
        custom_tooltip = {
            text = byz_mission_events.4.c.tt
            hidden_effect = {
                # Release Turkish Beyliks as separate vassals
                if = {
                    limit = {
                        region:anatolia_region = {
                            any_location_in_region = {
                                owner ?= root
                                culture = culture:turkish_culture
                            }
                        }
                    }
                    region:anatolia_region = {
                        ordered_location_in_region = {
                            limit = {
                                owner ?= root
                                culture = culture:turkish_culture
                                development >= 15
                            }
                            order_by = development
                            max = 2
                            create_vassal = {
                                culture = culture:turkish_culture
                                religion = religion:sunni
                                government_type = government_type:monarchy
                                name = "Beylik"
                            }
                        }
                    }
                }
            }
        }
        
        # Improve relations with Christian neighbors
        every_country = {
            limit = {
                religion_group = religion_group:christian
                is_neighbor_of = root
            }
            add_opinion = {
                target = root
                modifier = opinion_byzantine_victory
            }
        }
    }
}

# EVENT 5: "Antioch Reclaimed" (Syrian Reconquest)

byz_mission_events.5 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.5.t
    desc = byz_mission_events.5.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = cultural
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:burghers_estate
            background = estate_type:clergy_estate
        }
        
        location:antioch = {
            save_scope_as = antioch_location
        }
        
        # Check religious composition
        scope:antioch_location = {
            save_scope_as = antioch_religion_scope
        }
        
        # Check for Muslim neighbors
        any_neighbor_country = {
            limit = {
                religion_group = religion_group:muslim
            }
            save_scope_as = muslim_neighbor
        }
    }

    # Option A: Restore the Patriarchate of Antioch (Religious)
    option = {
        name = byz_mission_events.5.a
        historical_option = yes
        
        add_legitimacy = 25
        add_religious_influence = religious_influence_severe_bonus
        
        custom_tooltip = {
            text = byz_mission_events.5.a.tt
            hidden_effect = {
                scope:antioch_location = {
                    # Build great cathedral
                    if = {
                        limit = {
                            has_building = building_type:cathedral
                        }
                        change_building_level_in_location = {
                            building = building_type:cathedral
                            value = 1
                        }
                    }
                    else = {
                        add_building_level = {
                            building = building_type:cathedral
                            levels = 2
                        }
                    }
                    
                    # Convert population gradually
                    every_pop = {
                        limit = {
                            owner = root
                            religion != religion:orthodox
                        }
                        split_pop = {
                            fraction = 0.3
                            religion = religion:orthodox
                        }
                    }
                    
                    add_location_modifier = {
                        modifier = antioch_patriarchate_restored
                        years = 25
                        mode = add
                    }
                }
            }
        }
        
        # Improve relations with other Orthodox countries
        every_country = {
            limit = {
                religion = religion:orthodox
            }
            add_opinion = {
                target = root
                modifier = opinion_orthodox_holy_site_recovered
            }
        }
    }

    # Option B: Make Antioch a Trade Hub (Economic)
    option = {
        name = byz_mission_events.5.b
        
        custom_tooltip = {
            text = byz_mission_events.5.b.tt
            hidden_effect = {
                scope:antioch_location = {
                    # Build trade buildings
                    if = {
                        limit = {
                            NOT = { has_building = building_type:marketplace }
                        }
                        add_building_level = {
                            building = building_type:marketplace
                            levels = 2
                        }
                    }
                    
                    if = {
                        limit = {
                            has_advance = entrepot_advance
                            NOT = { has_building = building_type:entrepot }
                        }
                        add_building_level = {
                            building = building_type:entrepot
                            levels = 1
                        }
                    }
                    
                    add_location_modifier = {
                        modifier = antioch_silk_road_hub
                        years = 20
                        mode = add
                    }
                }
                
                # Create trade route to Constantinople
                capital.market = {
                    add_temporary_trade_route = {
                        from = scope:antioch_location
                        to = capital
                        goods = goods:silk
                        months = 120
                    }
                }
            }
        }
        
        # Economic bonuses
        add_gold = {
            value = monthly_income_trade_and_tax
            multiply = 8
        }
        
        # Boost burghers
        add_estate_satisfaction = {
            type = estate_type:burghers_estate
            value = estate_satisfaction_mild_bonus
        }
    }

    # Option C: Fortress of the East (Military)
    option = {
        name = byz_mission_events.5.c
        
        add_army_tradition = army_tradition_mild_bonus
        
        custom_tooltip = {
            text = byz_mission_events.5.c.tt

                        limit = {
                            has_advance = fortress_advance
                        }

                            building = building_type:fortress
                            levels = 1
                        }
                    }
                    else_if = {
                        limit = {
                            has_advance = bastion_advance
                        }
                        add_building_level = {
                            building = building_type:bastion
                            levels = 2
                        }
                    }
                    
                    # Garrison bonus
                    add_location_modifier = {
                        modifier = eastern_bulwark
                        years = 20
                        mode = add
                    }
                }

                if = {
                    limit = {
                        exists = scope:muslim_neighbor
                    }
                    scope:muslim_neighbor = {
                        region:levant_region = {
                            every_location_in_region = {
                                limit = {

# Event for when multiple major missions are completed
byz_mission_events.100 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.100.t
    desc = byz_mission_events.100.d
    
    trigger = {
        OR = {
            # Completed Jerusalem + Antioch + Alexandria
            AND = {
                has_variable = jerusalem_liberated_flag
                has_variable = antioch_liberated_flag
                has_variable = alexandria_restored_flag
            }
            # Completed Pentarchy
            has_variable = pentarchy_restored_flag
            # Roman Empire restored
            has_variable = roman_empire_restored_flag
        }
        is_great_power = yes
        prestige >= 80
    }
    
    mean_time_to_happen = {
        months = 1
    }
    
    fire_only_once = yes
    
    illustration_tags = {
        10 = empire
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:crown_estate
            background = estate_type:nobles_estate
        }
        
        # Determine which achievement was reached
        if = {
            limit = {
                has_variable = roman_empire_restored_flag
            }
            set_variable = {
                name = greatest_achievement
                value = 3  # Roman Empire
            }
        }
        else_if = {
            limit = {
                has_variable = pentarchy_restored_flag
            }
            set_variable = {
                name = greatest_achievement
                value = 2  # Pentarchy
            }
        }
        else = {
            set_variable = {
                name = greatest_achievement
                value = 1  # Three Patriarchates
            }
        }
    }

    option = {
        name = byz_mission_events.100.a
        # Proclaim the Restoration
        
        add_prestige = prestige_severe_bonus
        add_legitimacy = 50
        
        custom_tooltip = {
            text = byz_mission_events.100.a.tt
            hidden_effect = {
                # Different effects based on achievement
                if = {
                    limit = {
                        var:greatest_achievement = 3
                    }
                    # Roman Empire: Permanent claims on entire Mediterranean
                    region:mediterranean_region = {
                        every_location_in_region = {
                            max = 20
                            add_permanent_claim = root
                        }
                    }
                }
                else_if = {
                    limit = {
                        var:greatest_achievement = 2
                    }
                    # Pentarchy: Massive religious influence
                    add_religious_influence = religious_influence_ultimate_bonus
                    every_country = {
                        limit = {
                            religion_group = religion_group:christian
                        }
                        add_opinion = {
                            target = root
                            modifier = opinion_christian_restoration
                        }
                    }
                }
                else = {
                    # Three Patriarchates: Development boost
                    every_owned_location = {
                        limit = {
                            has_building_with_at_least_one_level = cathedral
                        }
                        change_development = development_weak_bonus
                    }
                }
                
                # Permanent modifier for completing major mission chain
                add_country_modifier = {
                    modifier = byzantine_restoration_complete
                    years = -1
                    mode = add
                }
            }
        }
    }
}
