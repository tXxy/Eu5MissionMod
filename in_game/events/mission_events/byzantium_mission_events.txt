#====================================================================
# EVENT 1: "The Theodosian Walls Restored" (Secure Constantinople)
#====================================================================
byz_mission_events.1 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.1.t
    desc = byz_mission_events.1.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = defensive
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:nobles_estate
            background = estate_type:peasants_estate
        }
        
        capital ?= {
            save_scope_as = constantinople_location
        }
        
        ruler_or_regent ?= {
            save_scope_as = current_ruler
        }
    }

    # Option A: The Grand Restoration (Historical & Prestige)
    option = {
        name = byz_mission_events.1.a

        add_prestige = prestige_mild_bonus
        if = {
            limit = {
                has_scope = constantinople_location
            }
            scope:constantinople_location = {
                if = {
                    limit = {
                        NOT = { has_building = building_type:bastion }
                    }
                    construct_building = {
                        building_type = building_type:bastion
                        instant = yes
                    }
                }
                else_if = {
                    limit = {
                        root = { has_advance = castle_advance }
                        NOT = { has_building = building_type:castle }
                    }
                    construct_building = {
                        building_type = building_type:castle
                        instant = yes
                    }
                }
            }
        }
        if = {
            limit = {
                has_scope = current_ruler
            }
            add_stability = stability_mild_bonus
            if = {
                limit = {
                    scope:current_ruler = {
                        mil >= 4
                    }
                }
                add_mil = {
                    value = {
                        value = scope:current_ruler.mil
                        divide = 10
                        min = 0
                        max = 2
                    }
                }
            }
        }
        else = {
            add_stability = stability_mild_bonus
        }
        
        set_country_flag = walls_restored_grandly
    }

    # Option B: Practical Repairs (Cheaper, Less Impressive)
    option = {
        name = byz_mission_events.1.b

        if = {
            limit = {
                has_scope = constantinople_location
            }
            scope:constantinople_location = {
                add_location_modifier = {
                    modifier = practical_fortifications
                    years = 15
                    mode = add
                }
            }
        }
        
        add_prestige = prestige_weak_penalty
        add_manpower = {
            value = {
                add = max_manpower
                multiply = 0.05
            }
        }
        
        set_country_flag = walls_restored_practically
    }

    # Option C: Reinforce the Sea Walls (Naval Focus)
    option = {
        name = byz_mission_events.1.c

        add_navy_tradition = navy_tradition_mild_bonus
        
        if = {
            limit = {
                has_scope = constantinople_location
            }
            scope:constantinople_location = {
                add_location_modifier = {
                    modifier = reinforced_sea_walls
                    years = 20
                    mode = add
                }
            }
        }
        
        change_societal_value = {
            type = land_vs_naval
            value = societal_value_minor_move_to_right
        }
        
        set_country_flag = sea_walls_enhanced
    }
}

# EVENT 2: "The Hexamilion Reconstructed" (Greek Defense Line)
#====================================================================
byz_mission_events.2 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.2.t
    desc = byz_mission_events.2.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = construction
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:peasants_estate
            background = estate_type:nobles_estate
        }
        
        location:corinth ?= {
            save_scope_as = corinth_location
        }
    }

    # Option A: The Historical Hexamilion (Defensive Focus)
    option = {
        name = byz_mission_events.2.a

        add_country_modifier = {
            modifier = hexamilion_complete
            years = 25
            mode = add
        }
        
        if = {
            limit = {
                has_scope = corinth_location
            }
            scope:corinth_location = {
                if = {
                    limit = {
                        NOT = { has_building = building_type:fortress }
                    }
                    construct_building = {
                        building_type = building_type:bastion
                        instant = yes
                    }
                }
            }
        }
        
        change_gold_effect = { scale = -5 }
        add_prestige = prestige_mild_bonus
        set_country_flag = hexamilion_restored_flag
    }

    # Option B: Fortified Isthmus (Network of Forts)
    option = {
        name = byz_mission_events.2.b

        add_country_modifier = {
            modifier = fortified_isthmus
            years = 20
            mode = add
        }
        
        area:morea_area = {
            if = {
                limit = {
                    any_location_in_area = {
                        owner ?= root
                        is_coastal = yes
                    }
                }
                ordered_location_in_area = {
                    limit = {
                        owner ?= root
                        is_coastal = yes
                    }
                    order_by = development
                    max = 3
                    if = {
                        limit = {
                            root = { has_advance = bastion_advance }
                            NOT = { has_building = building_type:bastion }
                        }
                        construct_building = {
                            building_type = building_type:bastion
                            instant = yes
                        }
                    }
                    else_if = {
                        limit = {
                            root = { has_advance = castle_advance }
                            NOT = { has_building = building_type:castle }
                        }
                        construct_building = {
                            building_type = building_type:castle
                            instant = yes
                        }
                    }
                }
            }
        }
        
        change_gold_effect = { scale = -8 }
        add_manpower = {
            value = {
                add = max_manpower
                multiply = 0.1
            }
        }
        set_country_flag = hexamilion_fortified_flag
    }

    # Option C: Watchtower Network (Cheaper, Detection Focus)
    option = {
        name = byz_mission_events.2.c

        add_country_modifier = {
            modifier = peloponnese_watchtowers
            years = 15
            mode = add
        }
        
        area:morea_area = {
            if = {
                limit = {
                    any_location_in_area = {
                        owner ?= root
                        topography = hills
                    }
                }
                ordered_location_in_area = {
                    limit = {
                        owner ?= root
                        topography = hills
                    }
                    order_by = development
                    max = 5
                    add_location_modifier = {
                        modifier = watchtower_network
                        years = 10
                        mode = add
                    }
                }
            }
        }
        
        change_gold_effect = { scale = -2 }
        add_army_tradition = army_tradition_weak_bonus
        set_country_flag = hexamilion_watchtowers_flag
    }

# EVENT 3: "Nicaea Liberated" (First Major Anatolian Victory)
#====================================================================
byz_mission_events.3 = {
    type = country_event
    category = mission_event
    title = byz_mission_events.3.t
    desc = byz_mission_events.3.d
    fire_only_once = yes
    
    illustration_tags = {
        10 = religious
        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:clergy_estate
            background = estate_type:peasants_estate
        }
        
        location:iznik ?= {
            save_scope_as = nicaea_location
            province = { save_scope_as = nicaea_province }
        }
        
        if = {
            limit = { country_exists = c:TUR }
            c:TUR = {
                save_scope_as = turkish_threat
            }
        }
    }

    # Option A: Re-establish the Patriarchate (Religious Focus)
    option = {
        name = byz_mission_events.3.a

        add_legitimacy = legitimacy_mild_bonus
        add_religious_influence = religious_influence_severe_bonus
        
        if = {
            limit = {
                has_scope = nicaea_location
            }
            scope:nicaea_location = {
                every_pop = {
                    limit = {
                        owner ?= root
                        religion != religion:orthodox
                    }
                    change_pop_religion = religion:orthodox
                }
                if = {
                    limit = {
                        NOT = { has_building = building_type:cathedral }
                    }
                    construct_building = {
                        building_type = building_type:cathedral
                        instant = yes
                    }
                }
                add_location_modifier = {
                    modifier = nicaea_patriarchate_restored
                    years = 20
                    mode = add
                }
            }
        }
        
        set_variable = {
            name = nicaea_council_planned
            value = 1
        }
        set_country_flag = nicaea_patriarchate_restored_flag
    }

    # Option B: Fortify the City (Military Focus)
    option = {
        name = byz_mission_events.3.b

        add_army_tradition = army_tradition_mild_bonus
        
        if = {
            limit = {
                has_scope = nicaea_location
            }
            scope:nicaea_location = {
                if = {
                    limit = {
                        root = { has_advance = castle_advance }
                    }
                    construct_building = {
                        building_type = building_type:castle
                        instant = yes
                    }
                }
                add_location_modifier = {
                    modifier = nicaea_fortified_bridgehead
                    years = 15
                    mode = add
                }
            }
        }
        if = {
            limit = {
                has_scope = turkish_threat
            }
            scope:turkish_threat = {
                area:marmara_area = {
                    every_location_in_area = {
                        limit = {
                            owner ?= scope:turkish_threat
                        }
                        max = 4
                        add_claim = root
                    }
                }
            }
        }
        
        add_estate_satisfaction = {
            type = estate_type:clergy_estate
            value = estate_satisfaction_weak_penalty
        }
        set_country_flag = nicaea_fortified_flag
    }

    # Option C: Re-settle with Greeks (Cultural Focus)
    option = {
        name = byz_mission_events.3.c

        culture = {
            add_cultural_influence = cultural_influence_mild_bonus
        }
        
        if = {
            limit = {
                has_scope = nicaea_location
            }
            add_migration = {
                owner = root
                from_location = capital
                to_location = scope:nicaea_location
                culture = culture:greek_culture
                amount = 0.08
                months = 60
                type = pop_type:burghers
            }
            scope:nicaea_location = {
                if = {
                    limit = {
                        NOT = { has_building = building_type:library }
                    }
                    construct_building = {
                        building_type = building_type:library
                        instant = yes
                    }
                }
                add_location_modifier = {
                    modifier = greek_resettlement_program
                    years = 12
                    mode = add
                }
                change_development = development_mild_bonus
            }
        }
        
        set_country_flag = nicaea_resettled_flag
    }
}

        10 = exterior
    }

    immediate = {
        event_illustration_estate_effect = {
            foreground = estate_type:nobles_estate
            background = estate_type:soldiers_estate
        }
        
        location:bursa ?= {
            save_scope_as = bursa_location
        }
        save_scope_as = byzantium_country
    }

    # Option A: The Final Campaign (If Ottomans destroyed)
    option = {
        name = byz_mission_events.4.a

        add_prestige = prestige_severe_bonus
        add_army_tradition = army_tradition_severe_bonus
        
        region:anatolia_region = {
            every_location_in_region = {
                limit = {
                    has_owner = yes
                    owner = {
                        culture = culture:turkish_culture
                    }
                }
                max = 8
                add_claim = root
            }
        }
        if = {
            limit = {
                has_scope = bursa_location
            }
            scope:bursa_location = {
                change_development = development_severe_bonus
                change_prosperity = prosperity_mild_bonus
            }
        }
        
        add_legitimacy = legitimacy_severe_bonus
        set_variable = {
            name = ottoman_spoils_available
            value = 1
        }
        set_country_flag = ottomans_destroyed_final
    }

    # Option B: The Humiliating Peace (If Ottomans vassalized)
    option = {
        name = byz_mission_events.4.b

        add_prestige = prestige_mild_bonus
        
        if = {
            limit = { country_exists = c:TUR }
            c:TUR = {
                save_scope_as = turkish_vassal
                add_liberty_desire = { value = -25 }
                if = {
                    limit = {
                        religion != religion:orthodox
                    }
                    add_opinion = {
                        modifier = opinion_forced_conversion
                        target = root
                        scale = 1
                    }
                }
                capital = {
                    province = {
                        every_location_in_province = {
                            add_claim = root
                        }
                    }
                }
            }
        }
        
        add_army_tradition = army_tradition_mild_bonus
        set_variable = {
            name = managing_turkish_vassal
            value = 1
        }
        set_country_flag = ottomans_vassalized_humiliated
    }

    # Option C: Divide and Rule (Diplomatic Approach)
    option = {
        name = byz_mission_events.4.c

        add_prestige = prestige_mild_bonus
        
        region:anatolia_region = {
            if = {
                limit = {
                    any_location_in_region = {
                        owner ?= root
                        culture = culture:turkish_culture
                    }
                }
                ordered_location_in_region = {
                    limit = {
                        owner ?= root
                        culture = culture:turkish_culture
                        development >= 15
                    }
                    order_by = development
                    max = 2
                    create_country_from_location = {
                        country_type = country_type:vassal
                        culture = culture:turkish_culture
                        religion = religion:sunni
                    }
                }
            }
        }
        
        every_country = {
            limit = {
                religion_group = religion_group:christian
                is_neighbor_of = root
            }
            add_opinion = {
                modifier = opinion_byzantine_victory
                target = root
                scale = 1
            }
        }
        set_country_flag = ottomans_divided_rule
    }
}

            AND = {

            has_country_flag = roman_empire_restored_flag
        }

    }
    
    mean_time_to_happen = {
        months = 1
    }
    
    illustration_tags = {
        10 = empire
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = {

            background = estate_type:nobles_estate
        }
        
        if = {
            limit = {
                has_country_flag = roman_empire_restored_flag
            }
            set_variable = {
                name = greatest_achievement
                value = 3
            }
        }
        else_if = {
            limit = {
                has_country_flag = pentarchy_supreme_authority_flag
            }
            set_variable = {
                name = greatest_achievement
                value = 2
            }
        }
        else = {
            set_variable = {
                name = greatest_achievement
                value = 1
            }
        }
    }

    option = {

        add_prestige = prestige_severe_bonus

        if = {
            limit = {

            }

                every_location_in_region = {
                    max = 20
                    add_claim = root
                }
            }
        }
        else_if = {
            limit = {
                var:greatest_achievement = 2
            }
            add_religious_influence = religious_influence_ultimate_bonus
            every_country = {
                limit = {
                    religion_group = religion_group:christian
                }

                    modifier = opinion_christian_restoration

                    scale = 1
                }
            }
        }
        else = {

                    has_building = building_type:cathedral
                }
                change_development = development_weak_bonus
            }
        }

            modifier = byzantine_restoration_complete
            years = -1
            mode = add
        }
    }
}
