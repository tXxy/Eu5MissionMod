# ==================== EVENT 1: THE VENETIAN LAGOON ====================
venice_mission_events.1 = {
	type = country_event
	category = mission_event
	title = venice_mission_events.1.t
	desc = venice_mission_events.1.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_poptype_effect = {
			foreground = pop_type:burghers
			background = pop_type:laborers
		}
		location:venezia = {
			save_scope_as = venice_location
		}
	}

	option = {
		name = venice_mission_events.1.a
		custom_tooltip = venice_mission_events.1.a.tt
		
		add_country_modifier = {
			modifier = ven_lagoon_consolidation
			years = 30
			mode = add
		}
	}

	option = {
		name = venice_mission_events.1.b
		custom_tooltip = venice_mission_events.1.b.tt
		
		location:venezia = {
			add_location_modifier = {
				modifier = ven_lagoon_fortifications
				years = 35
				mode = add
			}
			change_development = 3
		}
		
		add_manpower = {
			value = max_manpower
			multiply = 0.15
		}
	}

	option = {
		name = venice_mission_events.1.c
		custom_tooltip = venice_mission_events.1.c.tt
		
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_severe_bonus
		}
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 6
		}
	}
}

# ==================== EVENT 2: THE DALMATIAN COAST ====================
venice_mission_events.2 = {
	type = country_event
	category = mission_event
	title = venice_mission_events.2.t
	desc = venice_mission_events.2.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_poptype_effect = {
			foreground = pop_type:mariners
			background = pop_type:burghers
		}
		location:zadar = {
			save_scope_as = dalmatian_location
		}
		location:split = {
			save_scope_as = dalmatian_location2
		}
	}

	option = {
		name = venice_mission_events.2.a
		custom_tooltip = venice_mission_events.2.a.tt
		
		add_country_modifier = {
			modifier = ven_dalmatian_coast
			years = 30
			mode = add
		}
		
		area:dalmacija_area = {
			every_location_in_area = {
				limit = { owner ?= root }
				change_prosperity = prosperity_mild_bonus
			}
		}
	}

	option = {
		name = venice_mission_events.2.b
		custom_tooltip = venice_mission_events.2.b.tt
		
		scope:dalmatian_location = {
			if = {
				limit = { NOT = { has_building_with_at_least_one_level = dock } }
				construct_building = {
					building_type = building_type:dock
					cost_multiplier = 0.4
					cost_multiplier_reason = generic_cost_reduction_build_tt
					instant = yes
				}
			}
			add_location_modifier = {
				modifier = ven_adriatic_harbor
				years = 25
				mode = add
			}
		}
		
		add_navy_tradition = 10
		add_sailors = {
			value = max_sailors
			multiply = 0.15
		}
	}

	option = {
		name = venice_mission_events.2.c
		custom_tooltip = venice_mission_events.2.c.tt
		
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		
		every_subject = {
			limit = { region:balkan_region = { any_location_in_region = { owner ?= prev } } }
			add_liberty_desire = -15
		}
	}
}

# ==================== EVENT 7: THE BATTLE OF LEPANTO VETERANS ====================
venice_mission_events.7 = {
	type = country_event
	category = mission_event
	title = venice_mission_events.7.t
	desc = venice_mission_events.7.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = armed
		10 = maritime
	}

	immediate = {
		event_illustration_poptype_effect = {
			foreground = pop_type:mariners
			background = pop_type:soldiers
		}
	}

	option = {
		name = venice_mission_events.7.a
		custom_tooltip = venice_mission_events.7.a.tt
		
		add_navy_tradition = 20
		add_army_tradition = 20
	}

	option = {
		name = venice_mission_events.7.b
		custom_tooltip = venice_mission_events.7.b.tt
				
		create_character = {
			first_name = name_sebastiano
			last_name = Venier
			culture = culture:venetian
			religion = religion:catholic
			adm = { 60 75 }
			dip = { 50 65 }
			mil = { 85 95 }
			add_trait = trait:naval_showman
			age = 65
			save_scope_as = lepanto_admiral
		}
		if = {
			limit = { exists = scope:lepanto_admiral }
			scope:lepanto_admiral = {
				move_country = root
			}
		}
	}

	option = {
		name = venice_mission_events.7.c
		custom_tooltip = venice_mission_events.7.c.tt
		
		add_country_modifier = {
			modifier = ven_holy_league_heroes
			years = 30
			mode = add
		}
		
		add_prestige = 10
		
		if = {
			limit = { country_exists = c:PAP }
			c:PAP ?= {
				add_opinion = {
					target = root
					modifier = opinion_holy_league_victory
				}
			}
		}
		
		if = {
			limit = { country_exists = c:CAS }
			c:CAS ?= {
				add_opinion = {
					target = root
					modifier = opinion_holy_league_victory
				}
			}
		}
	}
}
