# ==================== EVENT 1: THE PRIVILEGIUM MAIUS ====================
# Fired by mission_aus_rudolf_the_founder
austria_mission_events.1 = {
	type = country_event
	category = mission_event
	title = austria_mission_events.1.t
	desc = austria_mission_events.1.d
	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:nobles_estate
		}
	}

	option = {
		name = austria_mission_events.1.a
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
		add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_bonus }
		add_country_modifier = { modifier = aus_palatine_archdukes years = 20 mode = add }
	}

	option = {
		name = austria_mission_events.1.b
		add_stability = stability_mild_bonus
		add_political_influence = political_influence_mild_bonus
	}
}
