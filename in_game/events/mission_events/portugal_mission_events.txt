portugal_missions_events.9 = { # The Sack of Thatta
	type = country_event
	title = portugal_missions_events.9.t
	desc = portugal_missions_events.9.d

	image = "gfx/interface/illustrations/event/backgrounds/special/devastation.dds"

	fire_only_once = yes

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	option = {
		name = portugal_missions_events.9.a
		
		add_gold = { # Sack the City
			value = root.country_tax_base
			multiply = 1.2
		}
		location:thatta = {
			every_pop = {
				add_pop_size = {
					value = pop_size
					multiply = -0.1
				}
			}
			change_prosperity = prosperity_extreme_penalty
			ordered_buildings_in_location = {
				order_by = building_level
				max = {
					value = 1
					add = {
						value = num_buildings
						divide = 4
					}
				}
				location = { catastrophic_destroy_building_levels_in_location = yes }
			}
		}
		add_government_power = government_power_mild_penalty

		historical_option = yes
	}

	option = { # Keep our troops in check
		name = portugal_missions_events.9.b

		add_gold = { 
			value = root.country_tax_base
			multiply = -0.5
		}
		add_government_power = government_power_extreme_bonus
		add_prestige = prestige_mild_bonus
	}
}
