# ==================== EVENT 1: SECURE THE ENGLISH CROWN ====================
england_mission_events.1 = {
	type = country_event
	category = mission_event
	title = england_mission_events.1.t
	desc = england_mission_events.1.d
	
	fire_only_once = yes
	
	illustration_tags = {
		10 = crown
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:nobles_estate 
			background = estate_type:nobles_estate 
		}
		
		ruler_or_regent ?= {
			save_scope_as = current_ruler
		}
	}

	option = {
		name = england_mission_events.1.a
		
		custom_tooltip = england_mission_events.1.a.tt
		
		add_country_modifier = {
			modifier = plantagenet_authority
			years = 15
			mode = add
		}
		add_gold = 150
				
		if = {
			limit = { exists = scope:current_ruler }
			scope:current_ruler = {
				add_character_modifier = {
					modifier = secure_throne
					years = 10
					mode = add
				}
			}
		}
	}

	option = {
		name = england_mission_events.1.b
		
		custom_tooltip = england_mission_events.1.b.tt
		
		add_country_modifier = {
			modifier = royal_prerogative
			years = 15

		}

# ==================== EVENT 2: DEVELOP LONDON ====================
england_mission_events.2 = {
	type = country_event
	category = mission_event
	title = england_mission_events.2.t
	desc = england_mission_events.2.d
	
	fire_only_once = yes
	
	illustration_tags = {
		10 = trade
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:burghers_estate 
			background = estate_type:burghers_estate 
		}
		
		capital ?= {
			save_scope_as = london_location
			market ?= {
				save_scope_as = london_market
			}
		}
	}

	option = {
		name = england_mission_events.2.a
		
		custom_tooltip = england_mission_events.2.a.tt
		
		if = {
			limit = { exists = scope:london_location }
			scope:london_location = {
				add_location_modifier = {
					modifier = london_commercial_hub
					years = 25
					mode = add
				}
				change_development = development_mild_bonus
			}
		}
		
		if = {
			limit = { exists = scope:london_market }
			scope:london_market = {
				add_merchant_power = {
					country = root
					key = london_trade_expansion
					years = 10
					power = merchant_power_mild_bonus
				}
			}
		}
	}

	option = {
		name = england_mission_events.2.b
		custom_tooltip = england_mission_events.2.b.tt
		
		if = {
			limit = { exists = scope:london_location }
			scope:london_location = {
				add_location_modifier = {
					modifier = london_cultural_center
					years = 25
					mode = add
				}
				change_development = development_weak_bonus
			}
		}
		
		culture = {
			add_cultural_tradition = cultural_tradition_severe_bonus
		}
		
		add_gold = {
			value = {
				add = monthly_income_trade_and_tax
				multiply = 4
			}
		}
		add_prestige = prestige_mild_bonus
	}
}

# ==================== EVENT 14: WIN HUNDRED YEARS WAR ====================
england_mission_events.14 = {
    type = country_event
    category = mission_event
    title = england_mission_events.14.t
    desc = england_mission_events.14.d
    
    fire_only_once = yes
    
    illustration_tags = {
        10 = empire
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { 
            foreground = estate_type:nobles_estate 
            background = estate_type:nobles_estate 
        }
        
        # Save important scopes
        c:FRA ?= {
            save_scope_as = defeated_france
        }
        
        # Check what we actually own
        set_local_variable = {
            name = holds_paris
            value = 0
        }
        if = {
            limit = { owns = location:paris }
            change_local_variable = {
                name = holds_paris
                value = 1
            }
        }
        
        set_local_variable = {
            name = holds_normandy
            value = 0
        }
        if = {
            limit = { area:normandy_area = { owner ?= root } }
            change_local_variable = {
                name = holds_normandy
                value = 1
            }
        }
        
        set_local_variable = {
            name = holds_aquitaine
            value = 0
        }
        if = {
            limit = { area:aquitaine_area = { owner ?= root } }
            change_local_variable = {
                name = holds_aquitaine
                value = 1
            }
        }
        
        # Check France's status

                name = france_is_subject

            name = france_is_union
            value = 0
        }
        if = {
            limit = {
                exists = scope:defeated_france

            }
            change_local_variable = {
                name = france_is_union
                value = 1
            }
        }
    }

    option = {
        name = england_mission_events.14.a
        custom_tooltip = england_mission_events.14.a.tt
        
        # Core Normandy if we hold it

            area:normandy_area = {
                every_location_in_area = {
                    limit = { owner ?= root }
                    add_core = root
                }
            }

        
        # Core Aquitaine if we hold it

            area:aquitaine_area = {
                every_location_in_area = {
                    limit = { owner ?= root }
                    add_core = root
                }
            }
        
        
        # Core Paris if we hold it

            location:paris = {
                province ?= {
                    every_location_in_province = {
                        limit = { owner ?= root }
                        add_core = root
                    }
                }
            }
        
        
        # Demand reparations if France exists
        if = {
            limit = {
                exists = scope:defeated_france
                local_var:france_is_subject = 0
                local_var:france_is_union = 0
            }
            scope:defeated_france = {
                add_gold = -750
                add_war_reparations = {
                    target = root
                    years = 10
                }
            }
            add_gold = 750
        }

            limit = {
                exists = scope:defeated_france
            }

                limit = { owner = scope:defeated_france }
                change_location_owner ?= root
                add_core = root

                limit = { owner = scope:defeated_france }
