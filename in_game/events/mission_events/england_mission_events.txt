# ==================== EVENT 1: SECURE THE ENGLISH CROWN ====================
england_mission_events.1 = {
    type = country_event
    category = mission_event
    title = england_mission_events.1.t
    desc = england_mission_events.1.d

    illustration_tags = {
        10 = crown

    }

    immediate = {
        event_illustration_estate_effect = { 
            foreground = estate_type:nobles_estate

            save_scope_as = current_ruler

        name = england_mission_events.1.a

            modifier = plantagenet_authority

            mode = replace

            limit = { exists = scope:current_ruler }

                    mode = replace

        }

        name = england_mission_events.1.b

# ==================== EVENT 11: FORM GREAT BRITAIN ====================
england_mission_events.11 = {
    type = country_event
    category = mission_event
    title = england_mission_events.11.t
    desc = england_mission_events.11.d
    
    fire_only_once = yes
    
    
    illustration_tags = {
        10 = empire
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { 
            foreground = estate_type:nobles_estate 
            background = estate_type:nobles_estate 
        }
        
        # Check if we can form GBR
        if = {
            limit = { tag = GBR }
            set_variable = {
                name = is_already_great_britain
                value = 1
            }
        }
    }

    option = {
        name = england_mission_events.11.a
        
        custom_tooltip = england_mission_events.11.a.tt
        
        add_country_modifier = {
            modifier = kingdom_of_great_britain
            years = -1
            mode = replace
        }
        
        add_prestige = prestige_extreme_bonus
        add_legitimacy = legitimacy_severe_bonus
        
        # If not already GBR, get formation bonuses
        if = {
            limit = { NOT = { has_variable = is_already_great_britain } }
            # Formation effects
            change_government_type = government_type:constitutional_monarchy
            
            # Unification bonuses
            every_owned_location = {
                limit = { region = region:great_britain_region }
                change_development = development_weak_bonus
            }
        }
    }

    option = {
        name = england_mission_events.11.b
        
        custom_tooltip = england_mission_events.11.b.tt
        
        add_country_modifier = {
            modifier = united_kingdom
            years = -1
            mode = replace
        }
        
        add_legitimacy = legitimacy_extreme_bonus
        
        # Focus on integration
        every_owned_location = {
            limit = {
                OR = {
                    culture = culture:scottish
                    culture = culture:welsh
                }
            }
            change_integration_progress = 30
        }
    }
}
