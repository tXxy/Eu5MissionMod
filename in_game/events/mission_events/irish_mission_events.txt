irish_mission_events.4 = {
	type = country_event
	category = mission_event
	title = irish_mission_events.4.t
	desc = irish_mission_events.4.d

	fire_only_once = yes

	illustration_tags = {
		10 = diplomacy
		10 = battle
	}

	# Conquest pad
	option = {
		name = irish_mission_events.4.a
		custom_tooltip = { text = ire_opens_conquest_branch_tt }
		add_country_modifier = {
			modifier = anti_english_policy
			years = -1
			mode = add
		}
	}

	# Diplomatiek pad
	option = {
		name = irish_mission_events.4.b
		custom_tooltip = { text = ire_opens_union_branch_tt }
		add_country_modifier = {
			modifier = irish_english_accord
			years = -1
			mode = add
		}
		add_prestige = 5
		add_legitimacy = 10

		if = {
			limit = {
				country_exists = c:PLE
				NOT = { c:PLE ?= { is_subject_of = root } }
			}
			add_casus_belli = {
				target = c:PLE
				type = casus_belli:cb_conquer_province
				years = 20
			}
		}
	}
}

# Senior Partner — Ierland kiest: join de unie of weiger
irish_mission_events.11 = {
	type = country_event
	category = mission_event
	title = irish_mission_events.11.t
	desc = irish_mission_events.11.d

	fire_only_once = yes

	# Accepteer de unie
	option = {
		name = irish_mission_events.11.a

		if = {
			limit = { country_exists = c:GBR }
			c:GBR = { create_union = root }
		}
		if = {
			limit = {
				NOT = { country_exists = c:GBR }
				country_exists = c:ENG
			}
			c:ENG = { create_union = root }
		}

		add_prestige = 25
		add_legitimacy = 25
		add_country_modifier = {
			modifier = gaelic_britannic_union
			years = -1
			mode = add
		}
	}

	# Weiger — claim + debuff
	option = {
		name = irish_mission_events.11.b

		if = {
			limit = { country_exists = c:GBR }
			add_casus_belli = {
				target = c:GBR
				type = casus_belli:cb_conquer_province
				years = 30
			}
			c:GBR = {
				add_opinion = { modifier = opinion_violated_sovereignty target = root scale = { value = 8 } }
				add_trust = { target = root value = -100 }
			}
		}
		if = {
			limit = {
				NOT = { country_exists = c:GBR }
				country_exists = c:ENG
			}
			add_casus_belli = {
				target = c:ENG
				type = casus_belli:cb_conquer_province
				years = 30
			}
			c:ENG = {
				add_opinion = { modifier = opinion_violated_sovereignty target = root scale = { value = 8 } }
				add_trust = { target = root value = -100 }
			}
		}

		add_prestige = 15
		add_stability = -15
		add_country_modifier = {
			modifier = anti_english_policy
			years = 20
			mode = add
		}
		add_country_modifier = {
			modifier = irish_broken_word
			years = -1
			mode = add
		}
	}
}

# What Does Rome Ask?
irish_mission_events.18 = {
	type = country_event
	category = mission_event
	title = irish_mission_events.18.t
	desc = irish_mission_events.18.d
	fire_only_once = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	# Betaal goud
	option = {
		name = irish_mission_events.18.a

		trigger = { gold >= 250 }

		add_gold = -250

		if = {
			limit = { country_exists = c:PAP }
			add_country_modifier = {
				modifier = irish_papal_protection
				years = 25
				mode = add
			}
			create_relation = {
				type = relation_type:guarantee
				first = c:PAP
				second = root
			}
		}
		if = {
			limit = { NOT = { country_exists = c:PAP } }
			add_country_modifier = {
				modifier = irish_self_reliance
				years = 20
				mode = add
			}
		}
	}

	# Bouw een kathedraal
	option = {
		name = irish_mission_events.18.b

		capital = {
			construct_building = {
				building_type = building_type:cathedral
				cost_multiplier = 0.0
				cost_multiplier_reason = generic_cost_reduction_build_tt
				instant = yes
			}
		}

		if = {
			limit = { country_exists = c:PAP }
			add_country_modifier = {
				modifier = irish_papal_protection
				years = 25
				mode = add
			}
			add_country_modifier = {
				modifier = irish_cathedral_built
				years = 20
				mode = add
			}
			create_relation = {
				type = relation_type:guarantee
				first = c:PAP
				second = root
			}
		}
		if = {
			limit = { NOT = { country_exists = c:PAP } }
			add_country_modifier = {
				modifier = irish_self_reliance
				years = 20
				mode = add
			}
		}
	}

	# Beloof een kruistocht
	option = {
		name = irish_mission_events.18.c

		if = {
			limit = { country_exists = c:PAP }
			add_country_modifier = {
				modifier = irish_crusade_pledge
				years = 15
				mode = add
			}
			create_relation = {
				type = relation_type:guarantee
				first = c:PAP
				second = root
			}
		}
		if = {
			limit = { NOT = { country_exists = c:PAP } }
			add_country_modifier = {
				modifier = irish_self_reliance
				years = 20
				mode = add
			}
		}
		if = {
			limit = {
				country_exists = c:TUR
				NOT = { c:ENG = { religion ?= religion:catholic } }
			}
			add_casus_belli = {
				target = c:TUR
				type = casus_belli:cb_conquer_province
				years = 20
			}
		}
	}

	# Weiger
	option = {
		name = irish_mission_events.18.e
		add_country_modifier = {
			modifier = irish_self_reliance
			years = -1
			mode = add
		}
		add_prestige = 5
	}
}
