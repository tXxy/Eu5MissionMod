# ==================== EVENT 1: THE CROWN OF NORWAY ====================
norway_mission_events.1 = {
	type = country_event
	category = mission_event
	title = norway_mission_events.1.t
	desc = norway_mission_events.1.d
	fire_only_once = yes
	is_triggered_only = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:nobles_estate
		}
	}

	option = {
		name = norway_mission_events.1.a
		# Lean into noble support — legitimacy but nobles get stronger
		add_legitimacy = 25
		add_stability = 5
		add_gold = 100
		add_country_modifier = {
			modifier = norwegian_noble_concessions
			years = 15
			mode = add
		}
	}

	option = {
		name = norway_mission_events.1.b
		# Lean into the church — stability and clergy power
		add_stability = 10
		add_legitimacy = 15
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
	}
}

# ==================== EVENT 31: THE OLD FAITH STIRS ====================
# The big choice: Lutheranism or a return to the Old Ways.
# CONFIRMED: Norse paganism's religion key in EU5 is "norse" (folk_european.txt), same
# family as romuva/sapmi_shamanism/muinaisusko already used in this mod's culture files.
norway_mission_events.31 = {
	type = country_event
	category = mission_event
	title = norway_mission_events.31.t
	desc = norway_mission_events.31.d
	fire_only_once = yes
	is_triggered_only = yes

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:clergy_estate
			background = estate_type:crown_estate
		}
	}

	option = {
		name = norway_mission_events.31.a
		# Embrace the Reformation
		change_religion = religion:lutheran
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:lutheran }
		add_stability = 5
		add_legitimacy = 5
		add_country_modifier = { modifier = chose_reformation_path years = -1 mode = add }
	}

	option = {
		name = norway_mission_events.31.b
		# Return to the Old Ways
		change_religion = religion:norse
		change_religion_for_ruler_and_family = { country = ROOT religion = religion:norse }
		add_stability = 5
		add_army_tradition = 5
		add_country_modifier = { modifier = chose_old_ways_path years = -1 mode = add }
	}

	option = {
		name = norway_mission_events.31.c
		# Stand firm with Rome — no religion change needed, Norway simply stays the course
		add_stability = 5
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
		add_country_modifier = { modifier = chose_catholic_path years = -1 mode = add }
	}
}

		name = norway_mission_events.37.b

		add_country_modifier = { modifier = nordic_cooperation years = 20 mode = add }

		c:DAN ?= { add_opinion = { modifier = opinion_demanded_additional_tribute target = root } }
