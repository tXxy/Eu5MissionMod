		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:nobles_estate

		name = ottoman_mission_events.1.a
		custom_tooltip = ottoman_mission_events.1.a.tt
		
		add_country_modifier = {
			modifier = tur_anatolian_unification
			years = 20
			mode = add_and_extend
		}
		
		area:anatolia_area = {
			ordered_location_in_area = {
				limit = { owner ?= root }
				order_by = development
				max = 8
				add_core = root

			}
		}
		
		add_legitimacy = 15
	}

	option = {
		name = ottoman_mission_events.1.b
		custom_tooltip = ottoman_mission_events.1.b.tt

ottoman_mission_events.4 = {
	type = country_event
	category = mission_event
	title = ottoman_mission_events.4.t
	desc = ottoman_mission_events.4.d
	fire_only_once = yes
	
	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:crown_estate
			background = estate_type:burghers_estate
		}
		location:constantinople = {
			save_scope_as = capital_location
		}
	}

	option = {
		name = ottoman_mission_events.4.a
		custom_tooltip = ottoman_mission_events.4.a.tt
		
		scope:capital_location = {
			add_location_modifier = {
				modifier = tur_imperial_capital
				years = -1
				mode = add
			}
			change_development = 8
			if = {
				limit = { NOT = { has_building_with_at_least_one_level = grand_palace } }
				construct_building = {
					building_type = building_type:grand_palace
					cost_multiplier = 0.5
					cost_multiplier_reason = generic_cost_reduction_build_tt
					instant = yes
				}
			}
			if = {
				limit = { NOT = { has_building_with_at_least_one_level = cathedral } }
				construct_building = {
					building_type = building_type:cathedral
					cost_multiplier = 0.5
					cost_multiplier_reason = generic_cost_reduction_build_tt
					instant = yes
				}
			}
		}
		
		add_prestige = 30
		add_stability = 20
		
		create_character = {
			first_name = Mimar
			last_name = Sinan
			adm = 85
			dip = 70
			mil = 50
			culture = culture:turkish
			religion = religion:sunni
			add_trait = trait:architect
			add_trait = trait:innovator
			age = 30
			save_scope_as = master_architect
			create_in_limbo = yes
		}
		
		if = {
			limit = { exists = scope:master_architect }
			scope:master_architect = {
				move_country = root
			}
		}
	}

	option = {
		name = ottoman_mission_events.4.b
		custom_tooltip = ottoman_mission_events.4.b.tt
		
		scope:capital_location = {
			add_location_modifier = {
				modifier = tur_constantinople_administrative
				years = -1
				mode = add
			}
			change_development = 5
		}
		
		add_country_modifier = {
			modifier = tur_administrative_capital
			years = 40
			mode = add_and_extend
		}
		
		add_gold = 500
	}
}
