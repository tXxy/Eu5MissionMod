# 26 fires if conditions worsen further (bloodbath).
# 29 fires as consequence of event 26 (aftermath).

# ==================== EVENT 28: UNREST IN SWEDEN ====================
# Fires if Denmark grows weak while Sweden is in the union.
denmark_mission_events.28 = {
	type = country_event
	category = mission_event
	title = denmark_mission_events.28.t
	desc = denmark_mission_events.28.d
	fire_only_once = yes

	trigger = {
		OR = { tag = DAN tag = SCA }
		in_union_with = c:SWE
		NOT = { has_country_modifier = bloodbath_infamy }
		OR = {
			stability <= 20
			legitimacy <= 40
			is_during_bankruptcy = yes
		}
	}

	mean_time_to_happen = {
		months = 24
	}

	illustration_tags = {
		10 = military
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:crown_estate
		}
	}

	option = {
		name = denmark_mission_events.28.a

		# Stabilize before the situation deteriorates further
		add_stability = 10
		add_legitimacy = 10
		c:SWE ?= {
			# Reduce Swedish antagonism by improving the opinion they have of us
			add_opinion = { modifier = opinion_opinion_influenced target = root }
		}
		# Flag that the warning has fired — gates the Bloodbath event
		set_variable = {
			name = kalmar_unrest_warning_fired
			value = 1
		}
	}
}

# ==================== EVENT 26: THE STOCKHOLM BLOODBATH ====================
# Fires if Denmark remains weak after event 28 already fired — the crisis escalates.
denmark_mission_events.26 = {
	type = country_event
	category = mission_event
	title = denmark_mission_events.26.t
	desc = denmark_mission_events.26.d
	fire_only_once = yes

	trigger = {
		OR = { tag = DAN tag = SCA }
		in_union_with = c:SWE
		NOT = { has_country_modifier = bloodbath_infamy }
		# Requires the unrest warning (event 28) to already have fired
		has_variable = kalmar_unrest_warning_fired
		OR = {
			stability <= 10
			legitimacy <= 30
			AND = {
				stability <= 20
				legitimacy <= 40
			}
		}
		c:SWE ?= {
			opinion = { target = root value <= 80 }
		}
	}

	mean_time_to_happen = {
		months = 48
	}

	illustration_tags = {
		10 = battle
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:nobles_estate
			background = estate_type:crown_estate
		}
	}

	option = {
		name = denmark_mission_events.26.a

		# The bloodbath poisons Swedish loyalty permanently
		add_country_modifier = {
			modifier = bloodbath_infamy
			years = 20
			mode = add
		}
		c:SWE ?= {
			add_country_modifier = {
				modifier = swedish_bloodbath_aftermath
				years = 15
				mode = add
			}
			# Drive Sweden toward independence via severe opinion crash
			add_opinion = { modifier = opinion_violated_sovereignty target = root }
		}
		add_prestige = -20
		add_legitimacy = -15

		trigger_event_non_silently = denmark_mission_events.29
	}
}
