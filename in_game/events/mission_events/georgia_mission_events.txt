# ==================== EVENT 1: UNITE GEORGIA ====================
georgian_mission_events.1 = {
	type = country_event
	category = mission_event
	title = georgian_mission_events.1.t
	desc = georgian_mission_events.1.d
	outcome = neutral
	fire_only_once = yes
	
	illustration_tags = {
		10 = happy
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:crown_estate 
			background = estate_type:nobles_estate 
		}
	}

	option = {
		name = georgian_mission_events.1.a
		custom_tooltip = georgian_mission_events.1.a.tt
		
		add_legitimacy = 25
		add_stability = 25
	}

	option = {
		name = georgian_mission_events.1.b
		custom_tooltip = georgian_mission_events.1.b.tt
		
		capital = {
			add_location_modifier = {
				modifier = geo_center_of_unification
				years = 15
				mode = add
			}
		}
	}
}
