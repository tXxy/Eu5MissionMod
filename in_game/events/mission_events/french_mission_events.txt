namespace = france_mission_events

# ==================== EVENT 1: SECURE THE FRENCH CROWN ====================
france_mission_events.1 = {
	type = country_event
	category = mission_event
	title = france_mission_events.1.t
	desc = france_mission_events.1.d
	picture = "gfx/interface/illustrations/royal_coronation"

	fire_only_once = yes

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:crown_estate 
			background = estate_type:nobles_estate 
		}
	}

	option = {
		name = france_mission_events.1.a
		custom_tooltip = france_mission_events.1.a.tt
		
		add_country_modifier = {
			modifier = fra_sacre_de_reims
			years = 30
			mode = add
		}
	}

	option = {
		name = france_mission_events.1.b
		custom_tooltip = france_mission_events.1.b.tt
		
		add_legitimacy = 30
		add_stability = 20
	}
}

	category = mission_event

	fire_only_once = yes

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:burghers_estate 
			background = estate_type:crown_estate 
		}
	}

	option = {
		name = france_mission_events.2.a
		custom_tooltip = france_mission_events.2.a.tt
		
		capital = {
			change_development = 4
			change_prosperity = prosperity_severe_bonus
		}
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 8
		}

		name = france_mission_events.2.b
		custom_tooltip = france_mission_events.2.b.tt
		
		add_country_modifier = {
			modifier = fra_parisian_culture
			years = 35
			mode = add
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_severe_bonus
		}
	}
}

# ==================== EVENT 3: LIBERATE NORMANDY ====================
france_mission_events.3 = {
	type = country_event
	category = mission_event
	title = france_mission_events.3.t
	desc = france_mission_events.3.d
	picture = "gfx/interface/illustrations/battle"

	fire_only_once = yes

	immediate = {
		event_illustration_poptype_effect = { 
			foreground = pop_type:soldiers 
			background = pop_type:nobles 
		}
	}

	option = {
		name = france_mission_events.3.a
		custom_tooltip = france_mission_events.3.a.tt
		
		add_prestige = 10
		
		if = {
			limit = { country_exists = c:ENG }
			add_casus_belli = {
				target = c:ENG
				type = casus_belli:cb_conquer_province
				months = 120
			}
		}
	}

	option = {
		name = france_mission_events.3.b
		custom_tooltip = france_mission_events.3.b.tt
		
		add_manpower = {
			value = max_manpower
			multiply = 0.15
		}
	}
}

	title = france_mission_events.9.t
	desc = france_mission_events.9.d
	picture = "gfx/interface/illustrations/golden_age"

	fire_only_once = yes

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:crown_estate 
			background = estate_type:burghers_estate 
		}
	}

	option = {
		name = france_mission_events.9.a
		custom_tooltip = france_mission_events.9.a.tt
		
		add_country_modifier = {
			modifier = fra_french_golden_age
			years = 60
			mode = add
		}
		
		culture:french = {
			add_cultural_tradition = 100
		}
		
		add_prestige = 250
		add_legitimacy = 150
	}

	option = {
		name = france_mission_events.9.b
		custom_tooltip = france_mission_events.9.b.tt
		
		ordered_owned_location = {
			limit = { development >= 30 }
			order_by = development
			max = 10
			change_development = 3
			change_prosperity = prosperity_extreme_bonus
		}
		
		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = 25
		}
	}
}
