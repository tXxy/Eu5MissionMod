# ==================== EVENT 1: CHOICE OF DYNASTY ==================== 
brandenburg_missions_events.1 = {
	type = country_event
	title = brandenburg_missions_events.1.t
	desc = brandenburg_missions_events.1.d
	outcome = neutral
	picture = "gfx/interface/illustrations/royal_succession"

	fire_only_once = yes

	immediate = {
		event_illustration_estate_effect = { 
			foreground = estate_type:nobles_estate 
			background = estate_type:nobles_estate 
		}
		
		save_scope_as = from_country
		create_character = {
			first_name = name_frederick
			dynasty = dynasty:hohenzollern_dynasty
			estate = estate_type:nobles_estate
			adm = { 60 90 }
			dip = { 60 90 }
			mil = { 60 90 }
			age = 30
			script = bra_friedrich_hohenzollern
			save_scope_as = hohenzollern_ruler
		}
		set_variable = { name = bra_blocked_from_hohenzoller_event value = yes }
	}

	# Option A: Get the Hohenzollern
	option = {
		name = brandenburg_missions_events.1.a
		custom_tooltip = brandenburg_missions_events.1.a.tt
		ai_will_select = { value = 100 }
		set_new_ruler = scope:hohenzollern_ruler
		add_legitimacy = -10
		add_stability = 10
	}

	# Option B: Status Quo
	option = {
		name = brandenburg_missions_events.1.b
		custom_tooltip = brandenburg_missions_events.1.b.tt
		ai_will_select = { value = 0 }
		add_prestige = 10
		add_stability = 15
	}
}

brandenburg_missions_events.3 = {
	type = country_event
	title = brandenburg_missions_events.3.t
	desc = brandenburg_missions_events.3.d
	outcome = neutral
	picture = "gfx/interface/illustrations/castle"

	# Option A: Set Berlin as Capital
	option = {
		name = brandenburg_missions_events.3.a
		custom_tooltip = brandenburg_missions_events.3.a.tt
		ai_will_select = { 
			value = 95 
			IF = {
				limit = {
					location:berlin = { development > root.capital.development }
				}
				multiply = 2
			}
		}
		
		set_capital = location:berlin 
		
		location:berlin = {
			add_location_modifier = {
				modifier = bra_construct_a_residency
				years = -1 
				mode = add_and_extend
			}
		}
	}

	# Option B: Status Quo
	option = {
		name = brandenburg_missions_events.3.b
		custom_tooltip = brandenburg_missions_events.3.b.tt
		ai_will_select = { value = 5 }
		
		capital = {
			add_location_modifier = {
				modifier = bra_construct_a_residency
				years = -1 
				mode = add_and_extend
			}
		}
	}
}

brandenburg_missions_events.11 = {

	desc = brandenburg_missions_events.11.desc
	outcome = neutral
	fire_only_once = yes
	
	illustration_tags = {
		10 = happy
 		10 = interior
	}

	immediate = {
		event_illustration_poptype_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

		every_country = {
			limit = { is_emperor = yes }
			save_scope_as = hre_emperor_nation
		}

		#Create Character Portraits in the Event
		scope:hre_emperor_nation.ruler = {
			save_scope_as = hre_emperor
		}
		root.ruler = {
			save_scope_as = player 
		}

		# Check the Influenced Princes 
		IF = {
			limit = { 
				c:RVB ?= {
					opinion = {
						target = root
						value >= 50
					}
				}
			}
			c:RVB ?= {
				add_to_list = influenced_princes_list
			}
		}
		IF = {
			limit = { 
				c:GMK ?= {
					opinion = {
						target = root
						value >= 550
					}
				}
			}
			c:GMK ?= {
				add_to_list = influenced_princes_list
			}
		}
		IF = {
			limit = { 
				c:MDN ?= {
					opinion = {
						target = root
						value >= 50
					}
				}
			}
			c:MDN ?= {
				add_to_list = influenced_princes_list
			}
		}
	}
	
	# Option A: Bribe Ravensberg
	option = {
		name = brandenburg_missions_events.11.a
		trigger = {
			any_in_list = {
				list = influenced_princes_list
				c:RVB ?= this
			}
		}
		add_gold = {
			value = c:RVB.country_tax_base
			multiply = -1
		}
		c:RVB ?= {
			make_subject_of = { target = root type = subject_type:vassal }
		}
	}

	# Optin B: Bribe the Mark
	option = {
		name = brandenburg_missions_events.11.b
		trigger = {
			any_in_list = {
				list = influenced_princes_list
				c:GMK ?= this
			}
		}
		add_gold = {
			value = c:GMK.country_tax_base
			multiply = -1
		}
		c:GMK ?= {
			make_subject_of = { target = root type = subject_type:vassal }
		}

	}

	# Option C: Bribe Minden
	option = {
		name = brandenburg_missions_events.11.c
		trigger = {
			any_in_list = {
				list = influenced_princes_list
				c:MDN ?= this
			}
		}
		add_gold = {
			value = c:MDN.country_tax_base
			multiply = -1
		}
		c:MDN ?= {
			make_subject_of = { target = root type = subject_type:vassal }
		}
	}

	# Option D: Bribe them All
	option = {
		name = brandenburg_missions_events.11.d
		trigger = {
			list_size = {
				name = influenced_princes_list
				value > 1
			}
		}
		add_gold = {
			value = root.country_tax_base
			multiply = -1.5
		}
		every_in_list = {
			list = influenced_princes_list 
			make_subject_of = { target = root type = subject_type:vassal }
		}
	}

	# Option E: Just Conquer a Foothold in Westphalia
	option = {
		name = brandenburg_missions_events.11.e
