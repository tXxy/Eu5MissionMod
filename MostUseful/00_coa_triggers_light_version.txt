# Overlords & Organizations

coa_def_ensign_trigger = {
	exists = scope:overlord
	scope:overlord = scope:custom_overlord
}

coa_def_colonial_nation_trigger = {
	scope:actor ?= {
		OR = {
			not = { scope_type = country }
			is_subject_type = colonial_nation
			is_subject_type = trade_company
			is_subject_type = conquistador
		}
	}
}

coa_def_capetian_trigger = {
	scope:actor ?= {
		ruler ?= {
			dynasty ?= {
				OR = {
					this = dynasty:burgundy_dynasty
					this = dynasty:dreux_dynasty
					this = dynasty:anjou_dynasty
					this = dynasty:artois_dynasty
					this = dynasty:bourbon_dynasty
					this = dynasty:capet_dynasty
					this = dynasty:de_evreux_dynasty
					this = dynasty:valois_dynasty
				}
			}
		}
	}
	coa_def_burgundy_portugal_trigger = no # Excluding main branch
}

coa_def_holland_ownership_trigger = {
	scope:actor ?= {
		OR = {
			owns = location:the_hague
			any_subject = { owns = location:the_hague }
		}	
		OR = {
			NOT = { country_exists = c:HOL }
			any_subject_or_below = { this = c:HOL }
		}
	}
}
